"""Compare retained failed forces to tighter CPU oracles, without retiming them."""

import hashlib
import importlib.metadata
import json
from pathlib import Path

import numpy as np
from pyscf import gto, scf


def main():
    prior = Path('/home/jzzeng/codes/vibeqc-issue409-packed/.artifacts/issue206')
    output = Path('.artifacts/issue206/cpu-force-diagnosis')
    output.mkdir(exist_ok=False)
    report = {
        'scope': __doc__,
        'cpu_only': True,
        'settings': {'energy_tolerance': 1e-14, 'gradient_tolerance': 1e-12,
                     'direct_scf_tolerance': 1e-14, 'max_cycles': 200},
        'versions': {name: importlib.metadata.version(name) for name in ('pyscf', 'numpy')},
        'runner_sha256': hashlib.sha256(Path(__file__).read_bytes()).hexdigest(),
        'cases': [],
    }

    def save():
        (output / 'report.json').write_text(json.dumps(report, indent=2) + '\n')

    def diagnose(label, path, geometries, samples, density_fitting):
        original = path.read_bytes()
        row = {'label': label, 'original_path': str(path),
               'original_sha256': hashlib.sha256(original).hexdigest(),
               'density_fitting': density_fitting, 'oracles': [], 'errors': {}}
        report['cases'].append(row)
        save()
        for slot, atoms in enumerate(geometries):
            mol = gto.M(atom=atoms, basis='def2-svp', unit='Bohr', cart=False,
                        charge=0, spin=0, verbose=0)
            engine = scf.RHF(mol)
            if density_fitting:
                engine = engine.density_fit(auxbasis='def2-svp')
            engine.conv_tol = 1e-14
            engine.conv_tol_grad = 1e-12
            engine.direct_scf_tol = 1e-14
            engine.max_cycle = 200
            energy = engine.kernel()
            gradient = engine.nuc_grad_method()
            if density_fitting:
                gradient.auxbasis_response = True
            forces = -gradient.kernel()
            residual = float(np.linalg.norm(engine.get_grad(engine.mo_coeff, engine.mo_occ)))
            row['oracles'].append({'slot': slot, 'energy': float(energy),
                                   'forces': forces.tolist(), 'converged': bool(engine.converged),
                                   'orbital_gradient_norm': residual})
            save()
            assert engine.converged and residual < 1e-11, (label, slot, residual)
            assert np.isfinite(energy) and np.isfinite(forces).all()
            print(label, slot, 'CPU gradient norm', residual, flush=True)
        oracle_e = np.array([x['energy'] for x in row['oracles']])
        oracle_f = np.array([x['forces'] for x in row['oracles']])
        for name, records in samples.items():
            row['errors'][name] = [
                {'maximum_energy_error': float(np.max(np.abs(np.array(x['energies']) - oracle_e))),
                 'maximum_force_error': float(np.max(np.abs(np.array(x['forces']) - oracle_f)))}
                for x in records
            ]
            print(label, name, 'maximum retained force error',
                  max(x['maximum_force_error'] for x in row['errors'][name]), flush=True)
        save()

    for batch in (1, 4):
        path = prior / f'final-direct/matrix/water-tetramer-def2-svp-spherical-b{batch}-none.json'
        data = json.loads(path.read_text())
        geometries = [[(a['element'], a['coordinates_bohr']) for a in g]
                      for g in data['workload']['geometries']]
        samples = {engine: [{'energies': x['energies_hartree'],
                             'forces': x['forces_hartree_per_bohr']}
                            for x in data[engine]['warm_samples']]
                   for engine in ('vibeqc', 'gpu4pyscf')}
        diagnose(f'direct-96-b{batch}', path, geometries, samples, False)

    path = prior / 'rebuild-smoke/changed/manifest.json'
    data = json.loads(path.read_text())
    first = data['samples'][0]
    geometries = [[(a['element'], r) for a, r in zip(original, target)]
                  for original, target in zip(first['workload']['geometries'],
                                              first['target_coordinates_bohr'])]
    assert all(x['target_coordinates_bohr'] == first['target_coordinates_bohr']
               for x in data['samples'])
    samples = {engine: [x for x in data['samples'] if x['engine'] == engine]
               for engine in ('vibeqc', 'gpu4pyscf')}
    diagnose('df-192-b1-changed', path, geometries, samples, True)
    report['status'] = 'completed CPU diagnosis; original failures remain unchanged'
    save()


if __name__ == '__main__':
    main()
