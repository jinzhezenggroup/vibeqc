"""Accuracy-only SCF-control probes; do not replace the original gate samples."""

import ctypes
import hashlib
import json
import os
from pathlib import Path

import cupy as cp
import numpy as np
from gpu4pyscf.lib import cutensor
from pyscf import gto, scf

from vibeqc import Calculator


def main():
    assert os.environ.get('SLURM_JOB_ID')
    assert cutensor.cutensor is not None and cutensor.contract_engine is None
    out = Path('.artifacts/issue206/convergence-controls-stock-v2')
    out.mkdir(exist_ok=False)
    prior = Path('/home/jzzeng/codes/vibeqc-issue409-packed/.artifacts/issue206')
    oracle_path = Path('.artifacts/issue206/cpu-force-diagnosis/report.json')
    oracle = json.loads(oracle_path.read_text())
    oracle_by_label = {x['label']: x for x in oracle['cases']}
    payload = {'scope': __doc__, 'runner_sha256': hashlib.sha256(Path(__file__).read_bytes()).hexdigest(), 'slurm_job_id': os.environ['SLURM_JOB_ID'],
               'cuda_visible_devices': os.environ.get('CUDA_VISIBLE_DEVICES'),
               'library_sha256': hashlib.sha256(Path(os.environ['VIBEQC_LIBRARY']).read_bytes()).hexdigest(),
               'oracle_sha256': hashlib.sha256(oracle_path.read_bytes()).hexdigest(),
               'controls': {'native_density_tolerances': [1e-10, 1e-11, 1e-12],
                            'stock_changed_gradient_tolerances': [1e-9, 1e-10, 1e-11]},
               'native': [], 'stock_changed': []}

    def save():
        (out / 'report.json').write_text(json.dumps(payload, indent=2) + '\n')

    def errors(energies, forces, label):
        reference = oracle_by_label[label]['oracles']
        return {'maximum_energy_error_vs_cpu': float(np.max(np.abs(np.asarray(energies) -
                    np.array([x['energy'] for x in reference])))),
                'maximum_force_error_vs_cpu': float(np.max(np.abs(np.asarray(forces) -
                    np.array([x['forces'] for x in reference]))))}

    save()
    for batch_size in ():
        path = prior / f'final-direct/matrix/water-tetramer-def2-svp-spherical-b{batch_size}-none.json'
        data = json.loads(path.read_text())
        atoms = [[(a['element'], a['coordinates_bohr']) for a in g] for g in data['workload']['geometries']]
        for tolerance in payload['controls']['native_density_tolerances']:
            calc = Calculator(method='rhf', basis='def2-svp', basis_representation='spherical',
                              device='cuda', density_fitting='none', max_iterations=100,
                              energy_tolerance=1e-12, density_tolerance=tolerance,
                              screening_tolerance=1e-14)
            calc._library.vibeqc_get_source_identity.restype = ctypes.c_char_p
            payload['native_identity'] = calc._library.vibeqc_get_source_identity().decode()
            row = {'batch': batch_size, 'density_tolerance': tolerance, 'samples': []}
            payload['native'].append(row)
            with calc.prepare_batch(atoms, charges=[0]*batch_size, multiplicities=[1]*batch_size,
                                    warm_start=True) as batch:
                batch.execute(strict=True)
                batch.set_warm_start_updates(False)
                batch.execute(strict=True)
                for _ in range(3):
                    result = batch.execute(strict=True)
                    forces = np.stack([x.forces for x in result.items])
                    row['samples'].append({'energies': result.energies.tolist(), 'forces': forces.tolist(),
                                           'iterations': [x.iterations for x in result.items],
                                           'density_rms': [x.density_rms for x in result.items],
                                           **errors(result.energies, forces, f'direct-96-b{batch_size}')})
                    save()
            print('native', batch_size, tolerance,
                  max(x['maximum_force_error_vs_cpu'] for x in row['samples']), flush=True)

    data = json.loads((prior / 'rebuild-smoke/changed/manifest.json').read_text())
    first = data['samples'][0]
    original = [(a['element'], a['coordinates_bohr']) for a in first['workload']['geometries'][0]]
    target = first['target_coordinates_bohr'][0]
    seed = cp.asarray(np.load(prior / 'rebuild-smoke/changed/gpu4pyscf-seed.npy', allow_pickle=False)[0])
    for tolerance in payload['controls']['stock_changed_gradient_tolerances']:
        mol = gto.M(atom=original, unit='Bohr', basis='def2-svp', cart=False, verbose=0)
        engine = scf.RHF(mol).density_fit(auxbasis='def2-svp').to_gpu()
        engine.conv_tol, engine.conv_tol_grad, engine.max_cycle = 1e-12, tolerance, 100
        engine.direct_scf_tol = 1e-14
        engine.kernel(dm0=seed.copy())
        engine.nuc_grad_method().kernel()
        assert engine.converged
        engine.reset(engine.mol.set_geom_(np.asarray(target), unit='Bohr', inplace=True))
        energy = engine.kernel(dm0=seed.copy())
        forces = cp.asnumpy(-engine.nuc_grad_method().kernel())
        assert engine.converged
        residual = float(cp.linalg.norm(engine.get_grad(engine.mo_coeff, engine.mo_occ)))
        row = {'gradient_tolerance': tolerance, 'energies': [float(energy)],
               'forces': [forces.tolist()], 'orbital_gradient_norm': residual,
               **errors([energy], [forces], 'df-192-b1-changed')}
        payload['stock_changed'].append(row)
        save()
        print('stock changed', tolerance, row['maximum_force_error_vs_cpu'], flush=True)
    payload['status'] = 'completed accuracy diagnostics; no replacement timing or gate verdict'
    save()


if __name__ == '__main__':
    main()
