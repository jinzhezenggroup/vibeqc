"""Observe native direct error mapping under GDB; intrusive, no timing claim."""
import hashlib
import json
import os
import pathlib
import subprocess

assert os.environ.get("SLURM_JOB_ID")
root=pathlib.Path.cwd()
out=root/".artifacts/issue206/debug-direct-192-v3"
lib=root/"build/cuda/libvibeqc.so"
env={k:v for k,v in os.environ.items() if not k.startswith("VIBEQC_")}
env.update(PYTHONPATH="python:.",OMP_NUM_THREADS="1",OPENBLAS_NUM_THREADS="1",VIBEQC_LIBRARY=str(lib),CUDA_LAUNCH_BLOCKING="1",LD_LIBRARY_PATH="/group/software/cuda-12.9.1/lib64:"+env.get("LD_LIBRARY_PATH",""))
cmd=["gdb","-batch","-x",str(out/"trace.gdb"),"--args","/tmp/vibeqc-308-gpu4pyscf-env/bin/python",".artifacts/issue206/diagnose-direct-192.py","--library",str(lib)]
payload={"slurm_job_id":os.environ["SLURM_JOB_ID"],"cuda_visible_devices":os.environ.get("CUDA_VISIBLE_DEVICES"),"command":cmd,"library_sha256":hashlib.sha256(lib.read_bytes()).hexdigest(),"controls":{k:v for k,v in env.items() if k.startswith("VIBEQC_") or k=="CUDA_LAUNCH_BLOCKING"}}
with (out/"gdb.log").open("x") as f:
 result=subprocess.run(cmd,env=env,stdout=f,stderr=subprocess.STDOUT,check=False)
payload["exit_code"]=result.returncode
(out/"manifest.json").write_text(json.dumps(payload,indent=2)+"\n")
print(json.dumps(payload),flush=True)
