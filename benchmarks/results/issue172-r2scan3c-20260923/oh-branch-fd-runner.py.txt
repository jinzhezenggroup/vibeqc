"""Source-pinned, branch-resolved OH force and total-energy FD diagnosis."""

import json
import os
from pathlib import Path

import numpy as np
from dftd4.interface import DampingParam, DispersionModel
from test_dft_complete_cpu import independent_semilocal_total_gradient
from tools.vibeqc_gcp.reference import evaluate_r2scan3c_gcp
from vibeqc import Calculator, KsOptions, load_r2scan3c_basis
from vibeqc._dft_gradient import StationaryKsState
from vibeqc_compiler.dft import NativeAO

BASE = [("O", (0.0, 0.0, 0.0)), ("H", (0.0, 0.0, 1.8))]
origin = np.asarray([position for _, position in BASE], dtype=np.float64)
target = origin.copy()
target[-1] += [0.002, -0.001, 0.003]
STEPS = (1e-3, 3e-4, 1e-4)
DIRECTIONS = {"transverse_y": (1, 1), "bond_z": (1, 2)}
OUT = Path(os.environ["VIBEQC_OH_BRANCH_FD_OUT"])
BASIS = load_r2scan3c_basis()
CALC = Calculator(
    method="r2scan-3c-uks",
    device="cuda",
    ks_options=KsOptions(),
    max_iterations=200,
    energy_tolerance=1e-12,
    density_tolerance=1e-10,
)
report = {
    "schema": "vibeqc.issue172.oh_branch_fd.v1",
    "source_tree": os.environ.get("VIBEQC_SOURCE_TREE"),
    "allocation": os.environ.get("VIBEQC_QZ_WORKLOAD"),
    "target_positions_bohr": target.tolist(),
    "steps_bohr": STEPS,
    "branches": {},
}


def save():
    pending = OUT.with_suffix(".json.tmp")
    pending.write_text(
        json.dumps(report, indent=2, default=str) + "\n", encoding="utf-8"
    )
    pending.replace(OUT)


def atoms_at(positions):
    return [
        (symbol, tuple(position))
        for (symbol, _), position in zip(BASE, positions, strict=True)
    ]


def state_at(batch, positions, *, oracle=False):
    with NativeAO(
        atoms_at(positions),
        basis=BASIS,
        representation="spherical",
        multiplicity=2,
    ) as basis:
        state = StationaryKsState.from_native(batch, basis)
        try:
            density = np.asarray(state.density).copy()
            coefficients = np.asarray(state.coefficients).copy()
            occupations = np.asarray(state.occupations).copy()
            orbital_energies = np.asarray(state.orbital_energies).copy()
            overlap = np.asarray(state.overlap).copy()
            info = {
                "identity": state.identity.to_payload(),
                "physical_residual": state.physical_residual,
                "occupied_indices": [
                    np.flatnonzero(occupation > 0.5).tolist()
                    for occupation in occupations
                ],
                "frontier_orbital_energies": [
                    orbital_energies[
                        spin,
                        max(0, int(np.count_nonzero(occ)) - 2) : int(
                            np.count_nonzero(occ)
                        )
                        + 2,
                    ].tolist()
                    for spin, occ in enumerate(occupations)
                ],
            }
            if oracle:
                electronic_energy, electronic_gradient = (
                    independent_semilocal_total_gradient(
                        basis,
                        state,
                        "r2scan-uks",
                        cart=False,
                        initial_guess="native",
                    )
                )
                numbers = np.asarray([8, 1], dtype=np.int32)
                damping = DampingParam(s6=1.0, s8=0.0, s9=2.0, a1=0.42, a2=5.65)
                d4 = DispersionModel(
                    numbers, positions, charge=0.0, ga=2.0, gc=1.0
                ).get_dispersion(damping, grad=True)
                gcp = evaluate_r2scan3c_gcp(
                    tuple(int(number) for number in numbers), positions
                )
                info["independent_state_seeded_total_energy"] = (
                    electronic_energy + float(d4["energy"]) + gcp.energy
                )
                info["independent_state_seeded_total_force"] = (
                    -electronic_gradient - d4["gradient"] - gcp.gradient
                ).tolist()
            return info, density, coefficients, occupations, overlap
        finally:
            state._source.close()


def item_record(item):
    return {
        "energy": item.energy,
        "forces": None if item.forces is None else item.forces.tolist(),
        "iterations": item.iterations,
        "physical_residual_rms": item.physical_residual_rms,
        "status": item.status,
    }


def run_branch(name):
    initial = origin if name == "prepared" else target
    with CALC.prepare_batch([atoms_at(initial)], multiplicities=[2]) as batch:
        first = batch.execute(strict=True, properties=("energy", "forces")).items[0]
        batch.set_warm_start_updates(False)
        if name == "prepared":
            for _ in range(3):
                batch.execute(strict=True, properties=("energy", "forces"))
            base = batch.execute(
                coordinates=[target], strict=True, properties=("energy", "forces")
            ).items[0]
        else:
            base = first
        base_info, base_density, base_coeff, base_occ, base_overlap = state_at(
            batch, target, oracle=True
        )
        row = {
            "initial_positions_bohr": initial.tolist(),
            "first": item_record(first),
            "target": item_record(base),
            "target_state": base_info,
            "finite_differences": [],
        }
        report["branches"][name] = row
        save()
        for direction_name, (atom, axis) in DIRECTIONS.items():
            analytic = -float(base.forces[atom, axis])
            for step in STEPS:
                samples = []
                for sign in (1, -1):
                    positions = target.copy()
                    positions[atom, axis] += sign * step
                    item = batch.execute(
                        coordinates=[positions],
                        strict=True,
                        properties=("energy",),
                    ).items[0]
                    info, density, _, _, _ = state_at(batch, positions)
                    samples.append(
                        {
                            "sign": sign,
                            "result": item_record(item),
                            "physical_residual": info["physical_residual"],
                            "occupied_indices": info["occupied_indices"],
                            "frontier_orbital_energies": info[
                                "frontier_orbital_energies"
                            ],
                            "density_max_delta_from_target_by_spin": [
                                float(np.max(np.abs(block)))
                                for block in density - base_density
                            ],
                        }
                    )
                estimate = (
                    samples[0]["result"]["energy"] - samples[1]["result"]["energy"]
                ) / (2 * step)
                row["finite_differences"].append(
                    {
                        "direction": direction_name,
                        "step_bohr": step,
                        "analytic_gradient": analytic,
                        "finite_difference": estimate,
                        "absolute_error": abs(estimate - analytic),
                        "samples": samples,
                    }
                )
                save()
                print(name, direction_name, step, abs(estimate - analytic), flush=True)
    return base_density, base_coeff, base_occ, base_overlap


save()
prepared = run_branch("prepared")
fresh = run_branch("fresh")
report["same_geometry_density_max_delta_by_spin"] = [
    float(np.max(np.abs(block))) for block in prepared[0] - fresh[0]
]
report["occupied_subspace_min_singular_by_spin"] = []
overlap = prepared[3][0] if prepared[3].ndim == 3 else prepared[3]
for spin in range(2):
    left = prepared[1][spin][:, prepared[2][spin] > 0.5]
    right = fresh[1][spin][:, fresh[2][spin] > 0.5]
    singular = np.linalg.svd(left.T @ overlap @ right, compute_uv=False)
    report["occupied_subspace_min_singular_by_spin"].append(float(np.min(singular)))
for name, row in report["branches"].items():
    reference = row["target_state"]
    row["independent_total_energy_error"] = abs(
        row["target"]["energy"] - reference["independent_state_seeded_total_energy"]
    )
    row["independent_total_force_max_error"] = float(
        np.max(
            np.abs(
                np.asarray(row["target"]["forces"])
                - np.asarray(reference["independent_state_seeded_total_force"])
            )
        )
    )
save()
print(
    "branch comparison complete",
    report["occupied_subspace_min_singular_by_spin"],
    flush=True,
)
