#!/usr/bin/env bash
set -uo pipefail
: "${SLURM_JOB_ID:?Run through finite Slurm allocation}"
root="$PWD/.artifacts/issue418"
output="$root/class-timing-v1"
mkdir "$output" || exit 2
printf '%s\n' "$SLURM_JOB_ID" > "$output/slurm-job.txt"
export LD_LIBRARY_PATH="$root/oracle:/group/software/cuda-12.9.1/lib64${LD_LIBRARY_PATH:+:$LD_LIBRARY_PATH}"
export OMP_NUM_THREADS=1 OPENBLAS_NUM_THREADS=1
python - <<'PY'
import json,subprocess
from pathlib import Path
rows=[]
for line in subprocess.check_output(['ps','-eo','pid=,ppid=,etimes=,comm=,args='],text=True).splitlines():
    pid,parent,age,name,args=line.split(None,4)
    if name not in ('nvcc','cicc','ptxas','nsys','ncu','compute-sanitizer'): continue
    # Two abandoned nsys control agents predate this campaign by 18 days.
    # They have no profiling target/controller and are not active captures.
    orphan=name=='nsys' and '--start-agent' in args and parent=='1' and int(age)>86400
    rows.append({'pid':int(pid),'parent':int(parent),'age_seconds':int(age),'name':name,'args':args,'orphan_control_agent':orphan})
Path('.artifacts/issue418/class-timing-v1/process-scan.json').write_text(json.dumps(rows,indent=2)+'\n')
raise SystemExit(1 if any(not r['orphan_control_agent'] for r in rows) else 0)
PY
if [[ "$?" != 0 ]]; then
  exit 3
fi
receipts=("$root/sanitizers-v1/"*.exit)
for receipt in "${receipts[@]}"; do
  if [[ "$(cat "$receipt")" != 0 ]]; then exit 4; fi
done
if [[ "${#receipts[@]}" != 9 ]]; then exit 5; fi
result=0
for phase in hot-objects-v2 rest-objects-v1; do
  "$root/$phase/benchmark" "$root/$phase/workload.txt" \
    > "$output/$phase.jsonl" 2> "$output/$phase.log"
  status=$?
  printf '%s\n' "$status" > "$output/$phase.exit"
  if [[ "$status" != 0 ]]; then result=1; fi
done
exit "$result"
