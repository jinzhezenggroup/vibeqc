"""Freeze the finished, committed combined library before GPU qualification."""

import hashlib
import json
import shutil
import subprocess
from pathlib import Path

from tools.benchmark_df_derivatives import source_identity

root=Path.cwd()
base='f5eb8dfb696ce4304c5151670c55fec78fdddb7d'
head=subprocess.check_output(['git','rev-parse','HEAD'],text=True).strip()
assert not subprocess.check_output(['git','status','--porcelain'],text=True).strip()
destination=root/'.artifacts/issue418/campaign'
destination.mkdir(exist_ok=False)
library=(root/'build/cuda/libvibeqc.so').resolve(strict=True)
shutil.copyfile(library,destination/'libvibeqc.so')
(destination/'libvibeqc.so.0').symlink_to('libvibeqc.so')
patch=subprocess.check_output(['git','diff','--binary',base,head,'--','src','include','python','tools/generate_rys_high_accuracy_tables.py'])
(destination/'source.patch').write_bytes(patch)
cache=root/'build/cuda/CMakeCache.txt'
shutil.copyfile(cache,destination/'CMakeCache.txt')
generated=root/'build/cuda/generated'
headers={str(p.relative_to(generated)):hashlib.sha256(p.read_bytes()).hexdigest() for p in sorted(generated.rglob('*')) if p.is_file() and p.suffix in ('.hpp','.cuh','.cu','.json')}
root_header=generated/'generated_df_rys.cuh'
qualified=root/'.artifacts/issue418/cuda-root-tests/df_rys_cuda0/generated_df_rys.cuh'
assert root_header.read_bytes()==qualified.read_bytes()
metadata={'base_commit':base,'head':head,'source_identity':source_identity(generated),'source_patch_sha256':hashlib.sha256(patch).hexdigest(),'library_sha256':hashlib.sha256((destination/'libvibeqc.so').read_bytes()).hexdigest(),'library_bytes':library.stat().st_size,'generated_hashes':headers,'root_header_matches_slurm_9891':True,'cmake_cache_sha256':hashlib.sha256(cache.read_bytes()).hexdigest(),'build_log_sha256':hashlib.sha256((root/'.artifacts/issue418/campaign-build.log').read_bytes()).hexdigest(),'toolchain':subprocess.check_output(['/group/software/cuda-12.9.1/bin/nvcc','--version'],text=True),'baseline_library':'/home/jzzeng/codes/vibeqc-issue206-direct-fix/.artifacts/issue206/aot-direct-matrix/libvibeqc.so','baseline_library_sha256':'7aed5066ca6de0ac8c6e700ab8571a7518f9ac88192b95b4517b2556ec24457f','baseline_library_bytes':206443312,'baseline_library_revision':'4635809a371bbb527debcec2584ea19610e84350','baseline_source_equivalent_to_f5eb8df':not subprocess.check_output(['git','diff','4635809a371bbb527debcec2584ea19610e84350',base,'--','src','include','python'])}
baseline=Path(metadata['baseline_library'])
assert hashlib.sha256(baseline.read_bytes()).hexdigest()==metadata['baseline_library_sha256']
metadata['library_growth_bytes']=metadata['library_bytes']-metadata['baseline_library_bytes']
(destination/'build.json').write_text(json.dumps(metadata,indent=2)+'\n')
print(json.dumps({k:v for k,v in metadata.items() if k!='generated_hashes'},indent=2))
