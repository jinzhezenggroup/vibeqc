#!/usr/bin/env bash
set -uo pipefail
: "${SLURM_JOB_ID:?Run through finite Slurm allocation}"
root="$PWD/.artifacts/issue418"
output="$root/sanitizers-v1"
mkdir "$output" || exit 2
printf '%s\n' "$SLURM_JOB_ID" > "$output/slurm-job.txt"
export LD_LIBRARY_PATH="$root/oracle:/group/software/cuda-12.9.1/lib64${LD_LIBRARY_PATH:+:$LD_LIBRARY_PATH}"
result=0
"$root/rest-objects-v1/benchmark" --qualify-only > "$output/rest-qualification.jsonl" 2> "$output/rest-qualification.log"
status=$?
printf '%s\n' "$status" > "$output/rest-qualification.exit"
if [[ "$status" != 0 ]]; then result=1; fi
for phase in hot-objects-v2 rest-objects-v1; do
  for check in memcheck racecheck initcheck synccheck; do
    /group/software/cuda-12.9.1/bin/compute-sanitizer --tool "$check" --error-exitcode 9 \
      "$root/$phase/benchmark" --qualify-only \
      > "$output/$phase-$check.stdout" 2> "$output/$phase-$check.stderr"
    status=$?
    printf '%s\n' "$status" > "$output/$phase-$check.exit"
    if [[ "$status" != 0 ]]; then result=1; fi
  done
done
exit "$result"
