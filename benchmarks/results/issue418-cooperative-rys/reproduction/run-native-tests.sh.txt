#!/usr/bin/env bash
set -uo pipefail
: "${SLURM_JOB_ID:?Run through finite Slurm allocation}"
output="$PWD/.artifacts/issue418/native-validation-v1"
mkdir "$output" || exit 2
printf '%s\n' "$SLURM_JOB_ID" > "$output/slurm-job.txt"
export LD_LIBRARY_PATH=/group/software/cuda-12.9.1/lib64${LD_LIBRARY_PATH:+:$LD_LIBRARY_PATH}
export VIBEQC_MP2_CUDA_TEST=1
ctest --test-dir build/cuda --output-on-failure > "$output/ctest.txt" 2>&1
status=$?
printf '%s\n' "$status" > "$output/ctest.exit"
exit "$status"
