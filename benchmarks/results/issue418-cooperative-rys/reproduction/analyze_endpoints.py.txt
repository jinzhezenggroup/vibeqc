"""Audit every frozen campaign sample without pooling intrusive diagnostics.

Retain all gates and raw durations, including negative results. Scientific work
must agree; resource metadata and implementation selection may differ, while
preparation duration is deliberately not interpreted as work.
"""

import hashlib
import json
import re
import statistics
from collections import Counter
from pathlib import Path

import numpy as np

ROOT = Path(__file__).resolve().parent
DIRECTORY = ROOT / "endpoints-v1"
BUILD = json.loads((ROOT / "campaign/build.json").read_text())


def journal_work(path):
    """Bind graph replay counts and explicit final work to the traced solve."""
    rows = [json.loads(line) for line in path.read_text().splitlines()]
    begin = Counter(row["name"] for row in rows if row["event"] == "BEGIN")
    values = {}
    for row in rows:
        if row["event"] == "VALUE":
            values.setdefault(row["key"], []).append(row["value"])
    keys = (
        "host_graph_replay", "device_iterations", "converged",
        "final_fock_evaluations", "final_eigen_solves", "final_density_updates",
        "final_candidate_rejections", "final_maximum_commutator",
        "final_density_rms", "final_maximum_idempotency_error",
    )
    return {"begins": dict(begin), "values": {k: values.get(k, []) for k in keys}}


def semantic_groups(sample):
    """Exclude timing/resources and the separately validated execution identity."""
    result = {}
    resource = re.compile(
        r"shell_\d{3}_(registers|.*shared_bytes|resident_.*limit|sm_thread_limit|rys_selected)$"
    )
    for group in sample["components"]["groups"]:
        result[(group["execution"], group["operation"])] = {
            "calls": group["calls"],
            "counters": {
                k: v for k, v in group["counter_sums"].items()
                if not resource.fullmatch(k) and k not in ("signature_packet_preparation_ns", "solve_epoch")
            },
            "logical_tiles": group["logical_tiles"],
            "repeated_tile_productions": group["repeated_tile_productions"],
        }
    return result


def analyze(path):
    payload = json.loads(path.read_text())
    samples = payload["samples"]
    diagnostics = payload.get("diagnostics", [])
    gates = {}
    gates["exit_zero"] = path.with_suffix(".exit").read_text().strip() == "0"
    gates["all_ten_clean_samples"] = len(samples) == 10 and all(
        s["scope"] == "clean endpoint" for s in samples
    )
    gates["each_pair_retained"] = Counter((s["policy"], s["repeat"]) for s in samples) == Counter(
        (p, r) for p in ("auto", "candidate") for r in range(5)
    )
    gates["frozen_library"] = payload["library_sha256"] == BUILD["library_sha256"]
    gates["frozen_native_identity"] = payload["native_source_identity"] == BUILD["native_source_identity"]
    gates["frozen_source_patch"] = payload["source_patch_sha256"] == BUILD["source_patch_sha256"]
    gates["all_energy_errors"] = all(s["maximum_energy_error"] <= 1e-9 for s in samples + diagnostics)
    gates["all_force_errors"] = all(
        s["maximum_force_error"] is None or s["maximum_force_error"] <= 1e-8
        for s in samples + diagnostics
    )
    gates["all_converged"] = all(c["converged"] for s in samples + diagnostics for c in s["convergence"])
    gates["fixed_iteration_work"] = len({tuple(s["iterations"]) for s in samples + diagnostics}) == 1
    gates["prime_iteration_work"] = all(s["prime_iterations"] == s["iterations"] for s in samples + diagnostics)
    metric = samples[0]["metric"]
    gates["metric_and_charged_memory_unchanged"] = all(s["metric"] == metric for s in samples + diagnostics)
    diag = {s["policy"]: s for s in diagnostics}
    gates["both_diagnostics"] = len(diag) == 2
    # Consecutive traced calls have different epochs by design. Each occupied
    # factor must refer to its own final determinant, never a preceding solve.
    gates["occupied_factor_matches_final_determinant"] = all(
        group["counter_sums"]["solve_epoch"] == next(
            row["value"] for row in sample["final_state_observations"] if row["key"] == "final_solve_epoch"
        ) and group["counter_sums"]["density_generation"] == next(
            row["value"] for row in sample["final_state_observations"] if row["key"] == "final_density_generation"
        )
        for sample in diagnostics for group in sample["components"]["groups"]
        if group["operation"] == "occupied_response_provenance"
    )
    gates["semantic_component_work_conserved"] = semantic_groups(diag["auto"]) == semantic_groups(diag["candidate"])
    journals = {
        p: journal_work(path.with_suffix(f".diagnostic-0-{p}.journal.jsonl"))
        for p in ("auto", "candidate")
    }
    gates["graph_replay_and_final_work_match"] = journals["auto"] == journals["candidate"]
    by_policy = {p: [s["seconds"] for s in samples if s["policy"] == p] for p in ("auto", "candidate")}
    medians = {p: statistics.median(v) for p, v in by_policy.items()}
    paired = [
        next(s["seconds"] for s in samples if s["policy"] == "auto" and s["repeat"] == r)
        - next(s["seconds"] for s in samples if s["policy"] == "candidate" and s["repeat"] == r)
        for r in range(5)
    ]
    shell_ms, classes, force_seconds = {}, {}, {}
    resource = {}
    for policy, sample in diag.items():
        groups = sample["components"]["groups"]
        force = next((g for g in groups if g["operation"] == "force_response"), None)
        shell_ms[policy] = sum(
            v for g in groups for k, v in g["gpu_exclusive_ms"].items()
            if re.fullmatch(r"shell_\d{3}_(packet|p\d+_\d+_\d+)", k)
        )
        classes[policy] = {} if force is None else {
            k: v for k, v in force["counter_sums"].items()
            if re.fullmatch(r"shell_\d{3}_(registers|.*shared_bytes|resident_.*limit|sm_thread_limit|rys_selected)", k)
        }
        force_seconds[policy] = sample.get("force_stage_seconds")
        resource[policy] = {"metric": sample["metric"], "process_device_resident_bytes": sample["process_device_resident_bytes"]}
    differences = {}
    for key in ("energies_hartree", "forces_hartree_per_bohr"):
        differences[key] = None if samples[0][key] is None else max(
            float(np.max(np.abs(np.asarray(s[key])-np.asarray(samples[0][key])))) for s in samples + diagnostics
        )
    return {
        "file": path.name, "sha256": hashlib.sha256(path.read_bytes()).hexdigest(),
        "gates": gates, "all_gates_passed": all(gates.values()),
        "clean_seconds": by_policy, "median_seconds": medians,
        "median_improvement_seconds": medians["auto"]-medians["candidate"],
        "relative_improvement": 1-medians["candidate"]/medians["auto"],
        "paired_improvement_seconds": paired,
        "maximum_energy_error": max(s["maximum_energy_error"] for s in samples + diagnostics),
        "maximum_force_error": max((s["maximum_force_error"] or 0 for s in samples + diagnostics), default=0),
        "maximum_cross_sample_difference": differences,
        "iterations": samples[0]["iterations"], "metric": metric,
        "warm_density_sha256": payload["warm_density_sha256"],
        "complete_shell_derivative_gpu_ms": shell_ms,
        "complete_force_stage_seconds": force_seconds,
        "diagnostic_work": journals, "class_resources": classes, "resources": resource,
    }


def main():
    records = [analyze(DIRECTORY / f"{aos}-{observable}.json") for aos in (96, 192, 384, 768) for observable in ("forces", "energy")]
    large = next(r for r in records if r["file"] == "768-forces.json")
    component_gain = 1-large["complete_shell_derivative_gpu_ms"]["candidate"]/large["complete_shell_derivative_gpu_ms"]["auto"]
    result = {
        "slurm": int((DIRECTORY / "slurm-job.txt").read_text()),
        "build": BUILD,
        "records": records,
        "all_sample_and_work_gates_passed": all(r["all_gates_passed"] for r in records),
        "complete_768_shell_derivative_reduction": component_gain,
        "predeclared_performance_gate_passed": large["median_improvement_seconds"] >= .20 or (
            component_gain >= .15 and large["median_improvement_seconds"] > 0
        ),
        "limits": "Warm frozen-density campaign only. One separate intrusive component sample per arm. Standalone baseline equivalence and final production source qualification remain separate.",
    }
    output = DIRECTORY / "analysis.json"
    output.write_text(json.dumps(result, indent=2) + "\n")
    for r in records:
        print(r["file"], r["median_seconds"], "improvement", r["relative_improvement"], "failed", [k for k,v in r["gates"].items() if not v])
    print("complete shell reduction",component_gain,"performance gate", result["predeclared_performance_gate_passed"])
    return 0 if result["all_sample_and_work_gates_passed"] and result["predeclared_performance_gate_passed"] else 1


if __name__ == "__main__":
    raise SystemExit(main())
