#!/usr/bin/env bash
# Independent old library verifies that the embedded auto arm preserved baseline
# science, work and charged memory. Single replays are not speedup evidence.
set -uo pipefail
: "${SLURM_JOB_ID:?Run inside a finite Slurm allocation}"
output_root="$PWD/.artifacts/issue418/baseline-equivalence-v1"
mkdir "$output_root" || exit 2
printf '%s\n' "$SLURM_JOB_ID" > "$output_root/slurm-job.txt"
export PYTHONPATH=python:. OMP_NUM_THREADS=1 OPENBLAS_NUM_THREADS=1
export LD_LIBRARY_PATH=/group/software/cuda-12.9.1/lib64${LD_LIBRARY_PATH:+:$LD_LIBRARY_PATH}
export VIBEQC_LIBRARY=/home/jzzeng/codes/vibeqc-issue206-direct-fix/.artifacts/issue206/aot-direct-matrix/libvibeqc.so
export VIBEQC_DF_REFERENCE_FINAL_VALIDATION=0 VIBEQC_DF_FINAL_PROJECTION=auto
export VIBEQC_DF_FORCE_SCREEN_ABS=off VIBEQC_DF_EXCHANGE=auto
export VIBEQC_DF_SEED_EXCHANGE=auto VIBEQC_DF_FINAL_EXCHANGE=auto
export VIBEQC_DF_RESIDENT_EXCHANGE=auto
# The old binary's source is byte-identical to the campaign base in these trees.
git diff --binary 4635809a371bbb527debcec2584ea19610e84350 f5eb8dfb696ce4304c5151670c55fec78fdddb7d -- src include python > "$output_root/source.patch"
[[ ! -s "$output_root/source.patch" ]] || exit 3
status=0
for aos in 384 768; do
  case "$aos" in
    384) case_name=water-hexadecamer-2s4-def2-svp-spherical ;;
    768) case_name=water-32mer-4s4-def2-svp-spherical ;;
  esac
  /tmp/vibeqc-pr-review-env/bin/python -m benchmarks.df_policy_endpoint \
    --aos "$aos" --output "$output_root/$aos-forces.json" --repeats 1 \
    --control VIBEQC_DF_SHELL_POLICY --policies auto --components-after \
    --source-patch "$output_root/source.patch" --skip-cold --expected-iterations 3 \
    --warm-checkpoint-in "/home/jzzeng/codes/vibeqc-issues388-391/.artifacts/issues388-391/final/ablations/$aos-VIBEQC_DF_DIIS_DOTS.checkpoint" \
    --reference "benchmarks/results/issue377-379-df/gpu4pyscf/$case_name.json" \
    > "$output_root/$aos-forces.stdout" 2> "$output_root/$aos-forces.stderr"
  rc=$?
  printf '%s\n' "$rc" > "$output_root/$aos-forces.exit"
  if [[ "$rc" != 0 ]]; then status=1; fi
done
exit "$status"
