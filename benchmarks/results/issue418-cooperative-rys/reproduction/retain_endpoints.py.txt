"""Append complete endpoint evidence without altering prior qualification bytes."""

import gzip
import hashlib
import json
from pathlib import Path

ROOT = Path(__file__).resolve().parent
DESTINATION = Path("benchmarks/results/issue418-cooperative-rys")
MANIFEST = DESTINATION / "manifest.json"
manifest = json.loads(MANIFEST.read_text())
indexed = {r["retained"]: r for r in manifest["records"]}


def retain(path, target, reason):
    """Retain original bytes, compressing large or whitespace-sensitive records."""
    raw = path.read_bytes()
    compressed = (
        len(raw) > 180000
        or any(line.rstrip() != line for line in raw.splitlines())
        or (bool(raw) and raw != raw.rstrip(b"\r\n") + b"\n")
    )
    target += ".gz" if compressed else ""
    encoded = gzip.compress(raw, compresslevel=9, mtime=0) if compressed else raw
    assert len(encoded) < 1024**2, target
    output = DESTINATION / target
    output.parent.mkdir(parents=True, exist_ok=True)
    if output.exists() and target not in indexed:
        raise RuntimeError(f"unindexed existing evidence: {target}")
    output.write_bytes(encoded)
    restored = gzip.decompress(output.read_bytes()) if compressed else output.read_bytes()
    assert restored == raw
    indexed[target] = {
        "source": str(path.relative_to(Path.cwd())) if path.is_relative_to(Path.cwd()) else str(path),
        "retained": target, "reason": reason,
        "original_bytes": len(raw), "original_sha256": hashlib.sha256(raw).hexdigest(),
        "stored_bytes": len(encoded), "stored_sha256": hashlib.sha256(encoded).hexdigest(),
        "encoding": "gzip" if compressed else "identity",
    }


for directory in ("endpoints-v1", "baseline-equivalence-v1", "candidate-holdouts-v1", "auxiliary-holdouts-v2", "native-validation-v1", "final-validation-v1"):
    source = ROOT / directory
    if not source.exists():
        continue
    if directory == "baseline-equivalence-v1" and not all((source / f"{aos}-forces.exit").exists() for aos in (384, 768)):
        continue
    if directory == "candidate-holdouts-v1" and not all((source / f"{group}.exit").exists() for group in ("shell", "replay", "auxiliary")):
        continue
    if directory == "final-validation-v1" and not (source / "analysis.json").exists():
        continue
    for path in sorted(source.iterdir()):
        if not path.is_file() or path.name.endswith(".source.patch"):
            continue
        if directory == "candidate-holdouts-v1" and path.suffix not in (".txt", ".exit", ".xml"):
            continue
        retain(path, f"endpoints/{directory}/{path.name}", "Complete campaign record; clean endpoints and intrusive diagnostics remain separate, all failures retained.")
for filename in ("build.json", "source.patch", "CMakeCache.txt"):
    retain(ROOT / "campaign" / filename, f"identity/campaign-{filename}", "Frozen campaign binary, source, generated inventory and exact build configuration.")
if (ROOT / "final-production/build.json").exists():
    for filename in ("build.json", "source.patch"):
        retain(ROOT / "final-production" / filename, f"identity/final-production-{filename}", "Final production binary and source selection with byte-identical generated mathematics.")
for filename in ("ci-python-failed-v1.log", "ownership-regression-v2.txt", "final-policy-host-tests.txt"):
    retain(ROOT / filename, f"qualification/{Path(filename).stem}.txt", "Original CI failure and repaired host regression retained: versioned CUDA ownership snapshot was stale after the native source edit.")
retain(ROOT / "final-qualification-protocol.json", "identity/final-qualification-protocol.json", "Predeclared final-default equivalence checks; smaller automatic domains retain their old behavior.")
for filename in ("freeze_campaign.py", "run-native-tests.sh", "run-endpoints.sh", "run-baseline-equivalence.sh", "run-candidate-holdouts.sh", "run-auxiliary-holdouts.sh", "analyze_endpoints.py", "analyze_baseline.py", "freeze_final.py", "run-final-validation.sh", "analyze_final.py", "retain_endpoints.py"):
    retain(ROOT / filename, f"reproduction/{filename}.txt", "Actual reproduction, qualification, analysis or byte-preserving retention driver.")
manifest["records"] = list(indexed.values())
manifest["status"] = "Campaign endpoint qualification retained; final production source/library qualification is recorded separately before promotion."
final_analysis = ROOT / "final-validation-v1/analysis.json"
if final_analysis.exists() and json.loads(final_analysis.read_text())["all_gates_passed"]:
    manifest["status"] = "All campaign, baseline-equivalence, integrated holdout and final production qualification gates passed. Automatic scope remains sm_120,384/768 AO with equal auxiliary spaces; keeps #206 open."
MANIFEST.write_text(json.dumps(manifest, indent=2) + "\n")
for row in manifest["records"]:
    stored = (DESTINATION / row["retained"]).read_bytes()
    assert hashlib.sha256(stored).hexdigest() == row["stored_sha256"]
    raw = gzip.decompress(stored) if row["encoding"] == "gzip" else stored
    assert len(raw) == row["original_bytes"] and hashlib.sha256(raw).hexdigest() == row["original_sha256"]
print(f"Verified all {len(indexed)} retained records")
