#!/usr/bin/env bash
# Exercise the integrated candidate with open shells, public representations,
# changed geometry, unequal auxiliary spaces and bounded response storage.
set -uo pipefail
: "${SLURM_JOB_ID:?Run inside a finite Slurm allocation}"
output="$PWD/.artifacts/issue418/candidate-holdouts-v1"
mkdir "$output" || exit 2
printf '%s\n' "$SLURM_JOB_ID" > "$output/slurm-job.txt"
export PYTHONPATH=python:. OMP_NUM_THREADS=1 OPENBLAS_NUM_THREADS=1
export LD_LIBRARY_PATH=/group/software/cuda-12.9.1/lib64${LD_LIBRARY_PATH:+:$LD_LIBRARY_PATH}
export VIBEQC_LIBRARY="$PWD/.artifacts/issue418/campaign/libvibeqc.so"
export VIBEQC_RESOURCE_CUDA_TEST=1 VIBEQC_DF_SHELL_POLICY=candidate
export VIBEQC_DF_WEIGHTED_EXECUTION=shell VIBEQC_DF_FORCE_SCREEN_ABS=off
status=0
for group in shell replay auxiliary; do
  case "$group" in
    shell) tests=(tests/python/test_df_shell_derivatives_cuda.py -k 'packet and compact and blas') ;;
    replay) tests=(tests/python/test_df_resident_response_cuda.py -k 'jk_scratch_survives_response_property_and_geometry_replays or jk_scratch_retains_discarded_metric_response') ;;
    auxiliary) tests=(tests/python/test_df_derivatives_cuda.py -k auxiliary_only_atom_hf_energy_derivative) ;;
  esac
  /tmp/vibeqc-pr-review-env/bin/python -m pytest -q "${tests[@]}" \
    --basetemp "$output/$group-tmp" --junitxml "$output/$group.xml" \
    > "$output/$group.txt" 2>&1
  rc=$?
  printf '%s\n' "$rc" > "$output/$group.exit"
  if [[ "$rc" != 0 ]]; then status=1; fi
done
exit "$status"
