"""Independent GPU4PySCF force-only and complete endpoints; finite Slurm only."""
import hashlib
import gpu4pyscf
import pyscf
import json
import os
from pathlib import Path
import subprocess
import sys
import time

import cupy as cp
import numpy as np
from gpu4pyscf import scf
from pyscf import gto

from benchmarks._cases import benchmark_cases
from benchmarks.compare_gpu4pyscf_batch import _gpu_sample, scaled_geometries
from benchmarks.df_policy_endpoint import CASES

assert os.environ.get('SLURM_JOB_ID')
aos = int(sys.argv[1])
profile = len(sys.argv) > 3 and sys.argv[3] == '--profile'
repeats = 1 if profile else 5
output = Path(sys.argv[2])
assert not output.exists()
case = benchmark_cases()[CASES[aos]]
refpath = Path('benchmarks/results/issue377-379-df/gpu4pyscf') / (CASES[aos]+'.json')
reference = json.loads(refpath.read_bytes())
atoms = scaled_geometries(case.atoms, 1)[0]
mol = gto.M(atom=atoms, basis=case.pyscf_basis, unit='Bohr', cart=False, verbose=0)
engine = scf.RHF(mol).density_fit(auxbasis=case.pyscf_basis)
engine.conv_tol = 1e-12
engine.conv_tol_grad = 1e-10
engine.direct_scf_tol = 1e-14
engine.max_cycle = 100
start = time.perf_counter()
engine.kernel()
force = -engine.nuc_grad_method().kernel()
cp.cuda.Stream.null.synchronize()
assert engine.converged
cold_seconds = time.perf_counter() - start
frozen = engine.make_rdm1().copy()
expected_energy = np.array(reference['gpu4pyscf']['energies_hartree'])
expected_force = np.array(reference['gpu4pyscf']['forces_hartree_per_bohr'])
payload = {'case': CASES[aos], 'slurm_job_id': os.environ['SLURM_JOB_ID'],
           'workload': reference['workload'], 'reference_sha256': hashlib.sha256(refpath.read_bytes()).hexdigest(),
           'runner_sha256': hashlib.sha256(Path(__file__).read_bytes()).hexdigest(),
           'versions': {'pyscf':pyscf.__version__, 'gpu4pyscf':gpu4pyscf.__version__, 'cupy':cp.__version__},
           'cold_seconds': cold_seconds, 'complete_samples': [], 'force_only_samples': [],
           'force_only_scope': 'Repeated complete analytic gradients returned to host from one converged state; SCF excluded.',
           'scope': 'intrusive profile' if profile else 'clean endpoint'}

def save():
    output.write_text(json.dumps(payload, indent=2)+'\n')

def check(sample):
    e = float(np.max(np.abs(np.asarray(sample['energies_hartree']) - expected_energy)))
    f = float(np.max(np.abs(np.asarray(sample['forces_hartree_per_bohr']) - expected_force)))
    sample.update(maximum_energy_error=e, maximum_force_error=f)
    save()
    assert e < 1e-9 and f < 1e-8, (e,f)

_gpu_sample([engine], [frozen], cp, -1)
if profile: cp.cuda.runtime.profilerStart()
for repeat in range(repeats):
    sample = _gpu_sample([engine], [frozen], cp, repeat)
    assert engine.converged
    payload['complete_samples'].append(sample)
    check(sample)
    print(aos, 'complete', repeat, sample['seconds'], flush=True)
# Warm gradient state exists before these timings begin; synchronize both edges.
for repeat in range(repeats):
    cp.cuda.Stream.null.synchronize()
    cp.cuda.nvtx.RangePush('gpu4pyscf/force-only')
    start = time.perf_counter()
    force = cp.asnumpy(-engine.nuc_grad_method().kernel())
    cp.cuda.Stream.null.synchronize()
    seconds = time.perf_counter()-start
    cp.cuda.nvtx.RangePop()
    sample = {'seconds':seconds, 'energies_hartree':[float(engine.e_tot)],
              'forces_hartree_per_bohr':[force.tolist()]}
    payload['force_only_samples'].append(sample)
    check(sample)
    print(aos, 'force', repeat, seconds, flush=True)
if profile: cp.cuda.runtime.profilerStop()
pool = cp.get_default_memory_pool()
payload['memory_after_warm'] = {'cupy_live_bytes':pool.used_bytes(), 'cupy_reserved_bytes':pool.total_bytes(),
    'nvidia_smi_processes': subprocess.check_output(['nvidia-smi','--query-compute-apps=pid,used_memory','--format=csv,noheader,nounits'],text=True),
    'pid':os.getpid(), 'scope':'Resident after warm; reserved pool is not a measured allocator high-water mark.'}
save()
