"""Fresh-owner cold and changed-geometry DF force endpoints under finite Slurm.

Cold means a fresh calculator/prepared molecular owner after common untimed
CUDA/library initialization. Changed calls start from a prepared original
geometry and one immutable validated checkpoint; the timed region contains the
complete geometry rebuild, normal SCF, final-state checks and full forces.
Neither mode includes process startup. Repeats retain different SCF branches.
"""
import argparse, ctypes, hashlib, json, os, time
from pathlib import Path
import numpy as np
from vibeqc import Calculator
from benchmarks._cases import benchmark_cases
from benchmarks.compare_gpu4pyscf_batch import convergence_payload, scaled_geometries
from benchmarks.df_policy_endpoint import CASES, independent_reference


def main():
    p=argparse.ArgumentParser(description=__doc__)
    p.add_argument('--aos',type=int,choices=CASES,required=True)
    p.add_argument('--phase',choices=('cold','changed'),required=True)
    p.add_argument('--checkpoint',type=Path)
    p.add_argument('--changed-reference',type=Path)
    p.add_argument('--output',type=Path,required=True)
    p.add_argument('--repeats',type=int,default=7)
    p.add_argument('--diagnostics-after',action='store_true',help='record separate intrusive setup/SCF/force traces after clean timing')
    a=p.parse_args()
    if not os.environ.get('SLURM_JOB_ID') or a.repeats<1:p.error('requires Slurm and positive repeats')
    if a.phase=='changed' and (not a.checkpoint or not a.changed_reference):p.error('changed phase requires checkpoint and independent reference')
    if a.output.exists():p.error('refusing to overwrite evidence')
    a.output.parent.mkdir(parents=True,exist_ok=True)
    case=benchmark_cases()[CASES[a.aos]]
    atoms=scaled_geometries(case.atoms,1)[0]
    original=np.array([r for _,r in atoms]);changed=original.copy();changed[1,0]+=.001
    reference_path=Path('benchmarks/results/issue377-379-df/gpu4pyscf')/(CASES[a.aos]+'.json')
    reference=json.loads(reference_path.read_text())
    changed_reference=json.loads(a.changed_reference.read_text()) if a.changed_reference else None
    if changed_reference:
        assert changed_reference['case']==CASES[a.aos]
        assert np.array_equal(changed_reference['coordinates_bohr'],changed)
        assert changed_reference['basis_representation']==case.basis_representation
        assert changed_reference['basis']==changed_reference['auxiliary_basis']==case.pyscf_basis
        assert changed_reference['converged']
    options=dict(method=case.method,basis=case.vibeqc_basis,basis_representation=case.basis_representation,
        device='cuda',density_fitting='cuda',auxiliary_basis=case.vibeqc_basis,
        density_fitting_memory_budget_bytes=0,screening_tolerance=1e-12,
        energy_tolerance=1e-12,density_tolerance=1e-10,max_iterations=100)
    lib=Calculator(**options)._library;lib.vibeqc_get_source_identity.restype=ctypes.c_char_p
    payload={'scope':__doc__,'phase':a.phase,'aos':a.aos,'case':CASES[a.aos],
        'slurm_job_id':os.environ['SLURM_JOB_ID'],'cuda_visible_devices':os.environ.get('CUDA_VISIBLE_DEVICES'),
        'native_identity':lib.vibeqc_get_source_identity().decode(),
        'library_sha256':hashlib.sha256(Path(os.environ['VIBEQC_LIBRARY']).read_bytes()).hexdigest(),
        'source_patch_sha256':hashlib.sha256(Path(os.environ['VIBEQC_LIBRARY']).with_name('source.patch').read_bytes()).hexdigest(),
        'runner_sha256':hashlib.sha256(Path(__file__).read_bytes()).hexdigest(),
        'reference_sha256':hashlib.sha256(reference_path.read_bytes()).hexdigest(),
        'coordinates_bohr':original.tolist(),'changed_coordinates_bohr':changed.tolist(),
        'controls':{k:v for k,v in os.environ.items() if k.startswith('VIBEQC_')},'samples':[],'diagnostics':[]}
    if a.checkpoint:payload['checkpoint_sha256']=hashlib.sha256(a.checkpoint.read_bytes()).hexdigest()
    if a.changed_reference:payload['changed_reference_sha256']=hashlib.sha256(a.changed_reference.read_bytes()).hexdigest()
    def save():a.output.write_text(json.dumps(payload,indent=2)+'\n')
    trace_keys=('VIBEQC_DF_TRACE','VIBEQC_DF_HOST_TRACE','VIBEQC_DF_PROGRESS_TRACE')
    assert not any(os.environ.get(key) for key in trace_keys),'clean timing requires traces disabled'
    def trace_paths(policy):
        return {key:a.output.with_suffix(f'.diagnostic-{policy}.{suffix}.jsonl')
            for key,suffix in zip(trace_keys,('cuda','host','journal'))}
    def traces(policy,enabled):
        for key,path in trace_paths(policy).items():
            if enabled:
                assert not path.exists(),'refusing to append to old diagnostic trace'
                os.environ[key]=str(path)
            else:os.environ.pop(key,None)
    runs=[(i,False) for i in range(a.repeats)]+([(0,True)] if a.diagnostics_after else [])
    def keep(result,batch,policy,repeat,seconds,diagnostic):
        item=result.items[0];forces=np.array([item.forces])
        if a.phase=='cold':
            energy,target=independent_reference(reference,a.aos,result.energies,forces,[item.basis_metadata])
        else:
            energy=np.array([changed_reference['energy_hartree']]);target=np.array([changed_reference['forces_hartree_per_bohr']])
        row={'policy':policy,'repeat':repeat,'seconds':seconds,'iterations':item.iterations,
            'convergence':convergence_payload(result),'energy_hartree':item.energy,
            'forces_hartree_per_bohr':item.forces.tolist(),
            'maximum_energy_error':float(np.max(np.abs(result.energies-energy))),
            'maximum_force_error':float(np.max(np.abs(forces-target))),
            'metric':[d.to_dict() for d in batch.last_density_fitting_metric_diagnostics()]}
        if diagnostic:
            from benchmarks.df_component_ledger import aggregate, read_trace
            row['components']=aggregate(read_trace(trace_paths(policy)['VIBEQC_DF_TRACE']))
            row['traces']={key:str(path) for key,path in trace_paths(policy).items()}
        payload['diagnostics' if diagnostic else 'samples'].append(row);save()
        print(a.phase,'diagnostic' if diagnostic else 'clean',repeat,policy,seconds,item.iterations,flush=True)
        assert item.executed_backend=='cuda' and np.isfinite(forces).all() and np.isfinite(item.energy)
        assert row['maximum_energy_error']<=1e-9 and row['maximum_force_error']<=1e-8
    try:
        os.environ['VIBEQC_DF_VALUE_STORAGE']='dense'
        if a.phase=='cold':
            # Initialize the common library/driver without retaining any model
            # owner. Both timed arms start with zero molecular caches.
            with Calculator(**options).prepare_batch([atoms]) as primer:
                primer.execute(strict=True)
            for repeat,diagnostic in runs:
                for policy in (('dense','packed') if repeat%2==0 else ('packed','dense')):
                    os.environ['VIBEQC_DF_VALUE_STORAGE']=policy
                    traces(policy,diagnostic)
                    start=time.perf_counter();calc=Calculator(**options)
                    with calc.prepare_batch([atoms]) as batch:
                        result=batch.execute(strict=True);seconds=time.perf_counter()-start
                        keep(result,batch,policy,repeat,seconds,diagnostic)
                    traces(policy,False)
        else:
            with Calculator(**options).prepare_batch([atoms]) as batch:
                payload['restore']=batch.load_checkpoint(a.checkpoint,allow_warm=True)
                batch.set_warm_start_updates(False)
                batch.execute(strict=True)
                from vibeqc.checkpoint import inspect_checkpoint
                verify_path=a.output.with_suffix('.frozen-input.checkpoint')
                def density_hashes():
                    batch.save_checkpoint(verify_path)
                    return {b['name']:b['sha256'] for b in inspect_checkpoint(verify_path).blobs if b['name'].startswith('density_')}
                frozen=density_hashes();payload['frozen_density_sha256']=frozen;assert frozen;save()
                for repeat,diagnostic in runs:
                    for policy in (('dense','packed') if repeat%2==0 else ('packed','dense')):
                        os.environ['VIBEQC_DF_VALUE_STORAGE']=policy
                        # Restore source geometry outside the changed timer;
                        # freeze D across both physical geometry transitions.
                        batch.execute([original],strict=True)
                        traces(policy,diagnostic)
                        start=time.perf_counter();result=batch.execute([changed],strict=True);seconds=time.perf_counter()-start
                        keep(result,batch,policy,repeat,seconds,diagnostic)
                        traces(policy,False)
                        assert density_hashes()==frozen,'changed execution mutated the frozen starting density'
    except Exception as error:
        payload['failure']=repr(error);save();raise

if __name__=='__main__':main()
