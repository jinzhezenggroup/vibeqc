"""Rebuild formatted #417 source only after the clean timing campaign ends.

Invoke once under a finite Slurm allocation after inspecting every predecessor.
The measured frozen library is immutable. This creates a distinct final record;
it never overwrites the identities or test evidence of the measured candidate.
"""

from __future__ import annotations

import fcntl
import hashlib
import json
import os
import subprocess
import time
from pathlib import Path


def main():
    if not os.environ.get("SLURM_JOB_ID"):
        raise SystemExit("requires finite Slurm with the assigned GPU visibility")
    root = Path.cwd()
    out = root / ".artifacts/issue409/final-qualified"
    # This check is deliberately repeated inside the allocation. A queued
    # launcher is not proof that earlier clean jobs have actually finished.
    queue = subprocess.check_output(["squeue", "-h", "-o", "%i %T"], text=True)
    remaining = [
        line
        for line in queue.splitlines()
        if line.split()[0] in {"9845", "9846", "9847", "9848", "9853", "9855"}
    ]
    if remaining:
        raise SystemExit("clean campaign still has live jobs: " + repr(remaining))
    out.mkdir(exist_ok=False)
    env = {k: v for k, v in os.environ.items() if not k.startswith("VIBEQC_")}
    env.update(
        PYTHONPATH="python:.",
        OMP_NUM_THREADS="1",
        OPENBLAS_NUM_THREADS="1",
        LD_LIBRARY_PATH="/group/software/cuda-12.9.1/lib64:"
        + env.get("LD_LIBRARY_PATH", ""),
        VIBEQC_LIBRARY=str(root / "build/cuda/libvibeqc.so"),
        VIBEQC_RESOURCE_CUDA_TEST="1",
        VIBEQC_DF_DERIVATIVE_CUDA_TEST="1",
    )
    payload = {
        "scope": __doc__,
        "slurm_job_id": os.environ["SLURM_JOB_ID"],
        "cuda_visible_devices": os.environ.get("CUDA_VISIBLE_DEVICES"),
        "revision": subprocess.check_output(
            ["git", "rev-parse", "HEAD"], text=True
        ).strip(),
        "dirty_paths": subprocess.check_output(
            ["git", "status", "--short"], text=True
        ).splitlines(),
        "runner_sha256": hashlib.sha256(Path(__file__).read_bytes()).hexdigest(),
        "checks": [],
    }

    def save():
        (out / "manifest.json").write_text(json.dumps(payload, indent=2) + "\n")

    def run(name, command, *, overrides=None):
        check = {"name": name, "command": command, "status": "running"}
        payload["checks"].append(check)
        save()
        start = time.perf_counter()
        with (out / f"{name}.log").open("x") as log:
            result = subprocess.run(
                command,
                env=env | (overrides or {}),
                stdout=log,
                stderr=subprocess.STDOUT,
                check=False,
            )
        check.update(
            exit_code=result.returncode,
            elapsed_seconds=time.perf_counter() - start,
            status="passed" if result.returncode == 0 else "failed",
            log_sha256=hashlib.sha256((out / f"{name}.log").read_bytes()).hexdigest(),
        )
        save()
        print(name, result.returncode, flush=True)
        if result.returncode:
            raise SystemExit(result.returncode)

    save()
    with (root / "build/cuda/.issue409-build.lock").open("w") as lock:
        fcntl.flock(lock, fcntl.LOCK_EX | fcntl.LOCK_NB)
        run(
            "build",
            [
                "/usr/bin/time",
                "--verbose",
                "--output",
                str(out / "build-resources.txt"),
                "cmake",
                "--build",
                "build/cuda",
                "--target",
                "vibeqc",
                "vibeqc_density_fitting_tests",
                "vibeqc_df_occupied_response_tests",
                "-j",
                "2",
            ],
        )
    paths = [
        "build/cuda/libvibeqc.so",
        "build/cuda/vibeqc_density_fitting_tests",
        "build/cuda/vibeqc_df_occupied_response_tests",
        "build/cuda/generated/build_identity.hpp",
    ]
    payload["sha256"] = {
        name: hashlib.sha256((root / name).read_bytes()).hexdigest() for name in paths
    }
    payload["build_resource_scope"] = (
        "Incremental rebuild of the formatted publishing source with the existing "
        "CMake/cache configuration; not a clean-build or baseline compile-cost comparison."
    )
    payload["build_resource_sha256"] = hashlib.sha256(
        (out / "build-resources.txt").read_bytes()
    ).hexdigest()
    payload["library_bytes"] = (root / "build/cuda/libvibeqc.so").stat().st_size
    # Static fatbin inspection reads compiled resource declarations; it is not
    # a device occupancy measurement or another scientific execution.
    run(
        "compiled-kernel-resources",
        [
            "/group/software/cuda-12.9.1/bin/cuobjdump",
            "--dump-resource-usage",
            "--gpu-architecture",
            "sm_120",
            "build/cuda/libvibeqc.so",
        ],
    )
    # Inspect the identity in a child so the parent never retains a GPU context
    # alongside tests or the portable clean endpoint wrapper.
    identity = subprocess.check_output(
        [
            "/tmp/vibeqc-pr-review-env/bin/python",
            "-c",
            (
                "import ctypes,os; lib=ctypes.CDLL(os.environ['VIBEQC_LIBRARY']); "
                "lib.vibeqc_get_source_identity.restype=ctypes.c_char_p; "
                "print(lib.vibeqc_get_source_identity().decode())"
            ),
        ],
        env=env,
        text=True,
    ).strip()
    payload["native_identity"] = identity
    save()
    native = [
        ("density", "./build/cuda/vibeqc_density_fitting_tests"),
        ("response", "./build/cuda/vibeqc_df_occupied_response_tests"),
    ]
    for name, executable in native:
        run(name, [executable])
    python = "/tmp/vibeqc-pr-review-env/bin/python"
    run(
        "molecular",
        [
            python,
            "-m",
            "pytest",
            "-q",
            "tests/python/test_df_packed_values_cuda.py",
            "tests/python/test_df_derivatives_cuda.py::test_df_generated_failed_item_preserves_successful_neighbor",
        ],
        overrides={"VIBEQC_DF_VALUE_STORAGE": "packed"},
    )
    run(
        "resources", [python, "-m", "pytest", "-q", "tests/python/test_df_resources.py"]
    )
    sanitizer = "/group/software/cuda-12.9.1/bin/compute-sanitizer"
    for name, executable in native:
        for tool in (
            ("memcheck", "initcheck", "synccheck")
            if name == "response"
            else ("memcheck",)
        ):
            command = [sanitizer, "--tool", tool, "--error-exitcode", "1"]
            if tool == "memcheck":
                command += ["--leak-check", "full"]
            run(f"{name}-{tool}", command + [executable])
    run(
        "portable-warm-192",
        [
            python,
            "benchmarks/experiments/issue409-packed-values/run_endpoints.py",
            "--library",
            env["VIBEQC_LIBRARY"],
            "--aos",
            "192",
            "--output",
            str(out / "portable-warm-192"),
        ],
    )
    payload["status"] = "passed"
    save()


if __name__ == "__main__":
    main()
