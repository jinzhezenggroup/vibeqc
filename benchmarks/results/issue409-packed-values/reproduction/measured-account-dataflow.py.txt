"""Reconcile retained #409 storage shapes with executed logical work.

Shape arithmetic is not a device-memory observation or a hardware traffic count.
Native event intervals include stream idle time. Nested operations/byte counters
can overlap and are deliberately not summed into a complete endpoint total.
Warm traces can reuse a graph captured before tracing started. Eager counters
therefore exclude replay work; full logical J FLOPs are derived separately from
observed operator counts and the source's two matrix-vector contractions.
"""

from __future__ import annotations

import hashlib
import json
from pathlib import Path

ROOT = Path("benchmarks/results/issue409-packed-values")
LINEAR_WORK = {
    "coulomb_contraction_flops",
    "packed_unpack_elements",
    "packed_projection_launches",
    "occupied_projection_flops",
    "occupied_projection_products",
    "occupied_exchange_flops",
    "occupied_exchange_products",
}
TRANSFER_COUNTERS = {
    "host_to_device_bytes",
    "device_to_host_bytes",
    "tensor_host_to_device_bytes",
    "density_upload_bytes",
    "raw_value_upload_bytes",
    "response_host_to_device_bytes",
    "factor_validation_download_bytes",
    "final_state_download_bytes",
}


def read(path):
    return json.loads(path.read_text())


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def main():
    manifest = read(ROOT / "manifest.json")
    for record in manifest["files"]:
        path = ROOT / record["path"]
        assert sha(path) == record["sha256"], path
    summary = read(ROOT / "phasea-summary.json")
    result = {
        "scope": __doc__,
        "runner_sha256": sha(Path(__file__)),
        "source_summary_sha256": sha(ROOT / "phasea-summary.json"),
        "capacity_definition": "src/scf/df_value_storage.hpp::df_packed_value_capacity",
        "capacity_definition_sha256": sha(Path("src/scf/df_value_storage.hpp")),
        "capacity_scope": "Publishing source after verified clang-format-only change; explicit allocated tensor/scratch shapes. Other source/metric/SCF/library reservations remain in reported plan diagnostics.",
        "cells": [],
        "unmeasured": [
            "No full hardware DRAM or whole-endpoint H2D/D2H/D2D total is inferred from overlapping semantic counters; CUPTI reports remain required.",
            "Raw-integral recurrence FLOPs are not counted by these records; raw output bytes and producer calls are retained instead.",
            "BLAS FLOPs use the conventional 2mnk count, not hardware instruction counts or padded SIMT lane operations.",
            "Plan reservation/peak metadata is distinct from sampled process peaks and the separate native-allocation ledger.",
        ],
    }
    for n in (96, 192, 384, 768):
        path = ROOT / "warm" / f"{n}-forces.json"
        data = read(path)
        reconciled = next(
            r
            for r in summary["cells"]
            if r["source"].endswith(f"clean-warm/{n}-forces.json")
        )
        assert len(data["samples"]) == 14 and len(data["diagnostics"]) == 2
        for sample in data["diagnostics"]:
            policy = sample["policy"]
            metric = sample["metric"][0]
            basis = sample["convergence"][0]["basis_metadata"]
            rank = basis["orbital"]["electrons"]["nalpha"]
            a = metric["effective_rank"]
            assert a == n and len(sample["metric"]) == 1
            # These retained warm cells have matching untruncated orbital and
            # auxiliary bases. Unequal/truncated cases must not reuse this
            # shortcut when they are added to the campaign.
            assert (
                basis["orbital"]["mathematical_identity"]
                == basis["auxiliary"]["mathematical_identity"]
            )
            pairs, offdiagonal = n * (n + 1) // 2, n * (n - 1) // 2
            q = metric["auxiliary_tile"]
            work = next(
                row for row in reconciled["diagnostic_work"] if row["policy"] == policy
            )
            groups = sample["components"]["groups"]
            row = {
                "aos": n,
                "auxiliaries": a,
                "policy": policy,
                "occupied_rank_capacity": rank,
                "source": str(path),
                "source_sha256": sha(path),
                "diagonal_pairs": n,
                "offdiagonal_pairs": offdiagonal,
                "lower_pairs": pairs,
                "full_pairs": n * n,
                "full_over_packed_per_tensor": n * n / pairs,
                "logical_full_tensor_bytes": 8 * n * n * a,
                "logical_packed_tensor_bytes": 8 * pairs * a,
                "logical_work": work["logical_operators"],
                "observed_graph_replays": work["graph_replays"],
                "observed_eager_linear_counters": {},
                "graph_construction_linear_counters": {},
                "transfer_counters_by_operation": [],
                "reported_plan_reservation": metric,
            }
            for group in groups:
                # A capture need not occur during this diagnostic endpoint.
                # Keep construction observations separate even when present;
                # replay counts alone cannot identify arbitrary templates.
                target = (
                    row["observed_eager_linear_counters"]
                    if group["execution"] == "stream"
                    else row["graph_construction_linear_counters"]
                )
                for key, value in group["counter_sums"].items():
                    if key in LINEAR_WORK:
                        target[key] = target.get(key, 0) + value
                observed = {
                    k: v
                    for k, v in group["counter_sums"].items()
                    if k in TRANSFER_COUNTERS
                }
                if observed and group["execution"] == "stream":
                    row["transfer_counters_by_operation"].append(
                        {"operation": group["operation"], "counters": observed}
                    )
            response = next(
                g
                for g in groups
                if g["operation"] == "force_response" and g["execution"] == "stream"
            )
            row["response"] = {
                "calls": response["calls"],
                "host_completion_ms": response["host_completion_ms"],
                "gpu_event_inclusive_ms": response["gpu_inclusive_ms"],
                "counters": response["counter_sums"],
                "host_exclusive_ms": response["host_exclusive_ms"],
                "gpu_event_exclusive_ms": response["gpu_exclusive_ms"],
            }
            row["J_K_component_scopes"] = [
                {
                    key: group[key]
                    for key in (
                        "operation",
                        "execution",
                        "calls",
                        "host_completion_ms",
                        "gpu_inclusive_ms",
                        "host_exclusive_ms",
                        "gpu_exclusive_ms",
                    )
                }
                for group in groups
                if group["operation"] in ("ri_j", "ri_k", "ri_k_occupied")
            ]
            eager_j = work["eager_operations"]["ri_j"]
            logical_j = work["logical_operators"]["J"]
            assert logical_j == eager_j + work["graph_replays"]
            assert not metric["streamed"]
            stored_pairs = pairs if policy == "packed" else n * n
            j_per_call = 4 * stored_pairs * a
            row["derived_logical_J_flops"] = {
                "definition": "Two resident GEMVs, conventional 2mn each; excludes density packing/scattering and hardware padding.",
                "formula": "4 * stored_pairs * auxiliaries * logical_J_calls",
                "per_call": j_per_call,
                "eager": j_per_call * eager_j,
                "graph_replay": j_per_call * work["graph_replays"],
                "total": j_per_call * logical_j,
                "source": "src/scf/cuda/df_coulomb.cpp",
                "source_sha256": sha(Path("src/scf/cuda/df_coulomb.cpp")),
                "replay_counter_coverage": "Graph was cached before tracing; the replay contribution is source-derived, not an observed CUDA counter.",
            }
            if policy == "packed":
                assert (
                    row["observed_eager_linear_counters"]["coulomb_contraction_flops"]
                    == j_per_call * eager_j
                )
                panel = n * n * q
                projection = max(panel, n * rank * a)
                row["simultaneous_value_plan_bytes"] = {
                    "immutable_raw_A": 8 * pairs * a,
                    "immutable_whitened_B": 8 * pairs * a,
                    "U_or_dense_panel": 8 * projection,
                    "dense_panel_1": 8 * panel,
                    "dense_panel_2": 8 * panel,
                    "explicit_pair_map": 0,
                }
                row["pair_map_reason"] = (
                    "deterministic triangular address arithmetic; no global pair map"
                )
                row["value_and_three_scratch_bytes"] = sum(
                    row["simultaneous_value_plan_bytes"].values()
                )
                assert (
                    row["value_and_three_scratch_bytes"]
                    < metric["device_resident_bytes"]
                )
                borrowed = response["counter_sums"].get("response_borrowed_jk_bytes")
                if borrowed is not None:
                    assert borrowed == 8 * (projection + 2 * panel)
                row["logical_pack_density_offdiagonal_adds"] = (
                    offdiagonal * work["logical_operators"]["J"]
                )
                cold = ROOT / "rebuild" / f"{n}-cold.json"
                if cold.exists():
                    cold_sample = next(
                        s for s in read(cold)["diagnostics"] if s["policy"] == policy
                    )
                    producer = next(
                        g
                        for g in cold_sample["components"]["groups"]
                        if g["operation"] == "resident_three_center_materialization"
                        and g["execution"] == "stream"
                    )
                    assert producer["calls"] == 1
                    counters = producer["counter_sums"]
                    assert counters["packed_raw_generation_calls"] == n
                    assert (
                        counters["raw_value_bytes"]
                        == counters["resident_raw_bytes"]
                        == counters["resident_transformed_bytes"]
                        == 8 * pairs * a
                    )
                    row["cold_producer"] = {
                        "source": str(cold),
                        "source_sha256": sha(cold),
                        "observed": producer,
                        "logical_whitening_flops": 2 * pairs * a * a,
                    }
            else:
                row["dense_raw_storage_note"] = (
                    "Host raw ownership, a leased full raw scratch buffer, and streamed generation are distinct paths. Do not count a separate dense device A without observed ownership evidence."
                )
            result["cells"].append(row)
    output = Path(".artifacts/issue409/storage-dataflow.json")
    output.write_text(json.dumps(result, indent=2) + "\n")
    print(
        output,
        len(result["cells"]),
        "representation arms; remaining observations are explicit",
    )


if __name__ == "__main__":
    main()
