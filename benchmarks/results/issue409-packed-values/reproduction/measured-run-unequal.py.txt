"""Qualify physical unequal auxiliary cases before larger capacity evidence."""
import hashlib
import json
import os
from pathlib import Path
import subprocess

assert os.environ.get('SLURM_JOB_ID')
root=Path.cwd();out=root/'.artifacts/issue409/unequal-v2';out.mkdir(exist_ok=False)
env={k:v for k,v in os.environ.items() if not k.startswith('VIBEQC_') and k!='CONTRACT_ENGINE'}
env.update(PYTHONPATH='python:.',OMP_NUM_THREADS='1',OPENBLAS_NUM_THREADS='1',
    LD_LIBRARY_PATH='/tmp/vibeqc-308-gpu4pyscf-env/lib/python3.13/site-packages/cutensor/lib:/group/software/cuda-12.9.1/lib64:'+env.get('LD_LIBRARY_PATH',''),
    VIBEQC_LIBRARY=str(root/'.artifacts/issue409/stage2/frozen/libvibeqc.so'),
    VIBEQC_DF_REFERENCE_FINAL_VALIDATION='0',VIBEQC_DF_FINAL_PROJECTION='auto',
    VIBEQC_DF_FORCE_SCREEN_ABS='off',VIBEQC_DF_EXCHANGE='auto',
    VIBEQC_DF_SEED_EXCHANGE='auto',VIBEQC_DF_FINAL_EXCHANGE='auto',
    VIBEQC_DF_RESIDENT_EXCHANGE='auto',VIBEQC_DF_SHELL_POLICY='auto')
checks=[]
for case,budget in [('water-def2-svp-spherical',0),('water-tetramer-def2-svp-spherical',0),
        ('water-hexadecamer-2s4-def2-svp-spherical',8<<30)]:
    reference=out/f'{case}-reference.json'
    for mode in ('reference','measure'):
        name=f'{case}-{mode}'
        command=['/tmp/vibeqc-308-gpu4pyscf-env/bin/python',
            '.artifacts/issue409/stage2/unequal-endpoints.py','--case',case,'--mode',mode,
            '--budget',str(budget),'--repeats','7','--output',str(out/f'{name}.json')]
        if mode=='measure':command+=['--reference',str(reference)]
        print('starting',name,flush=True)
        with (out/f'{name}.log').open('x') as stream:
            run=subprocess.run(command,env=env,stdout=stream,stderr=subprocess.STDOUT)
        checks.append({'name':name,'command':command,'exit_code':run.returncode})
        (out/'manifest.json').write_text(json.dumps({'scope':__doc__,
            'slurm_job_id':os.environ['SLURM_JOB_ID'],'checks':checks,
            'runner_sha256':hashlib.sha256(Path(__file__).read_bytes()).hexdigest()},indent=2)+'\n')
        print(name,run.returncode,(out/f'{name}.log').read_text().splitlines()[-3:],flush=True)
        if run.returncode:raise SystemExit(run.returncode)
