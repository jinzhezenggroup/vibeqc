"""Intrusive constrained DF endpoint and native allocation observation.

The 32-GiB numeric ledger is an observation ceiling, not the small-AO Python
inventory or a whole-process memory guarantee. The separately supplied DF
budget is enforced by the actual native value/response planners.
"""
import argparse, ctypes, hashlib, json, os, resource, subprocess, threading, time
from pathlib import Path

assert os.environ.get('SLURM_JOB_ID')
import numpy as np
from vibeqc import Calculator
from vibeqc.resources import (CpuResourceObservation, ResourceBudget, ResourceCandidate,
    ResourceEstimate, ResourceIdentity, ResourceRequest, plan_resources)
from vibeqc.resources_native import NativeDeviceLedger
from benchmarks._cases import benchmark_cases
from benchmarks.compare_gpu4pyscf_batch import convergence_payload
from benchmarks.df_policy_endpoint import CASES, independent_reference
from benchmarks.df_component_ledger import aggregate, read_trace

p=argparse.ArgumentParser(description=__doc__)
p.add_argument('--aos',type=int,choices=(384,768),required=True)
p.add_argument('--storage',choices=('dense','packed'),required=True)
p.add_argument('--budget',type=int,required=True)
p.add_argument('--checkpoint',type=Path,required=True)
p.add_argument('--output',type=Path,required=True)
a=p.parse_args()
if a.output.exists() or a.budget<0:p.error('nonnegative budget and fresh output required')
a.output.parent.mkdir(parents=True,exist_ok=True)
os.environ['VIBEQC_DF_VALUE_STORAGE']=a.storage
case=benchmark_cases()[CASES[a.aos]]
reference_path=Path('benchmarks/results/issue377-379-df/gpu4pyscf')/(CASES[a.aos]+'.json')
reference=json.loads(reference_path.read_text())
calc=Calculator(method=case.method,basis=case.vibeqc_basis,basis_representation=case.basis_representation,
    device='cuda',density_fitting='cuda',auxiliary_basis=case.vibeqc_basis,
    density_fitting_memory_budget_bytes=a.budget,screening_tolerance=1e-12,
    energy_tolerance=1e-12,density_tolerance=1e-10,max_iterations=100)
lib=calc._library;lib.vibeqc_get_source_identity.restype=ctypes.c_char_p
identity=ResourceIdentity('hf','diagnostic-ledger','cuda','fp64',json.dumps({'aos':a.aos}),
    ('energy','forces'),'diagnostic ceiling; not inventory admission')
request=ResourceRequest('hf',identity,(ResourceCandidate('observe','resident',(
    ResourceEstimate('diagnostic numeric capacity ceiling',32<<30,'device:0',0,0),)),))
ledger=NativeDeviceLedger(lib,plan_resources([request],ResourceBudget()))
payload={'scope':__doc__,'storage':a.storage,'aos':a.aos,'df_budget_bytes':a.budget,
    'slurm_job_id':os.environ['SLURM_JOB_ID'],'cuda_visible_devices':os.environ.get('CUDA_VISIBLE_DEVICES'),
    'native_identity':lib.vibeqc_get_source_identity().decode(),
    'library_sha256':hashlib.sha256(Path(os.environ['VIBEQC_LIBRARY']).read_bytes()).hexdigest(),
    'checkpoint_sha256':hashlib.sha256(a.checkpoint.read_bytes()).hexdigest(),
    'reference_sha256':hashlib.sha256(reference_path.read_bytes()).hexdigest(),
    'runner_sha256':hashlib.sha256(Path(__file__).read_bytes()).hexdigest(),
    'controls':{k:v for k,v in os.environ.items() if k.startswith('VIBEQC_')}}
def save():a.output.write_text(json.dumps(payload,indent=2)+'\n')
def memory():
    raw=subprocess.check_output(['nvidia-smi','--query-compute-apps=pid,used_memory','--format=csv,noheader,nounits'],text=True)
    devices={int(pid):int(mib)*1024**2 for line in raw.splitlines() for pid,mib in [line.split(',')]}
    return [time.monotonic_ns(),devices[os.getpid()],int(Path('/proc/self/statm').read_text().split()[1])*os.sysconf('SC_PAGE_SIZE')]
try:
    with CpuResourceObservation(lib,ledger=ledger):
        with calc.prepare_batch([case.atoms]) as batch:
            payload['restore']=batch.load_checkpoint(a.checkpoint,allow_warm=True)
            batch.set_warm_start_updates(False)
            initial=batch.execute(strict=True)
            target_energy,target_force=independent_reference(reference,a.aos,initial.energies,
                np.array([i.forces for i in initial.items]),[i.basis_metadata for i in initial.items])
            payload['primed_ledger']=ledger.to_dict()
            payload['primed_convergence']=convergence_payload(initial)
            trace=a.output.with_suffix('.cuda.jsonl')
            os.environ['VIBEQC_DF_TRACE']=str(trace)
            os.environ['VIBEQC_DF_HOST_TRACE']=str(a.output.with_suffix('.host.jsonl'))
            os.environ['VIBEQC_DF_PROGRESS_TRACE']=str(a.output.with_suffix('.journal.jsonl'))
            samples=[];failures=[];stop=threading.Event()
            def sample():
                try:
                    while not stop.is_set():
                        samples.append(memory());stop.wait(.05)
                except Exception as error:failures.append(repr(error))
            before=memory();worker=threading.Thread(target=sample);worker.start()
            try:
                start=time.perf_counter();result=batch.execute(strict=True);seconds=time.perf_counter()-start
            finally:stop.set();worker.join()
            after=memory()
            energy_error=float(np.max(np.abs(result.energies-target_energy)))
            forces=np.array([i.forces for i in result.items]);force_error=float(np.max(np.abs(forces-target_force)))
            payload.update(intrusive_seconds=seconds,iterations=[i.iterations for i in result.items],
                convergence=convergence_payload(result),energies_hartree=result.energies.tolist(),
                forces_hartree_per_bohr=forces.tolist(),energy_error=energy_error,force_error=force_error,
                metric=[d.to_dict() for d in batch.last_density_fitting_metric_diagnostics()],
                components=aggregate(read_trace(trace)),observed_ledger=ledger.to_dict(),
                process_samples=samples,sampling_failures=failures,
                sampled_warm_device_peak=max(s[1] for s in [before,after]+samples),
                sampled_warm_host_peak=max(s[2] for s in [before,after]+samples),
                lifetime_host_peak_bytes=resource.getrusage(resource.RUSAGE_SELF).ru_maxrss*1024)
            save()
            assert np.isfinite(forces).all() and np.isfinite(result.energies).all()
            assert energy_error<=1e-9 and force_error<=1e-8
            assert samples and not failures
    payload['closed_ledger']=ledger.to_dict();save()
    assert payload['closed_ledger']['live_bytes']==0
except Exception as error:
    payload['failure']=repr(error);payload['failure_ledger']=ledger.to_dict();save();raise
finally:ledger.close()
print(a.storage,payload['intrusive_seconds'],payload['iterations'],payload['observed_ledger'],flush=True)
