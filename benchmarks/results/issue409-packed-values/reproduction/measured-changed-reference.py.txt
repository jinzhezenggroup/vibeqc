"""Independent stock GPU4PySCF full-force oracle for one displaced molecule."""
import argparse, hashlib, json, os
from pathlib import Path
import numpy as np
from benchmarks._cases import benchmark_cases
from benchmarks.compare_gpu4pyscf_batch import scaled_geometries
from benchmarks.df_policy_endpoint import CASES
p=argparse.ArgumentParser();p.add_argument('--aos',type=int,choices=CASES,required=True);p.add_argument('--output',type=Path,required=True);a=p.parse_args()
assert os.environ.get('SLURM_JOB_ID') and not a.output.exists()
import cupy as cp
import gpu4pyscf, pyscf
from pyscf import gto, scf
case=benchmark_cases()[CASES[a.aos]];atoms=scaled_geometries(case.atoms,1)[0]
positions=np.array([r for _,r in atoms]);positions[1,0]+=.001
mol=gto.M(atom=[(z,r) for (z,_),r in zip(atoms,positions)],unit='Bohr',basis=case.pyscf_basis,
    cart=case.basis_representation=='cartesian',spin=case.multiplicity-1,charge=case.charge,verbose=0)
engine=scf.RHF(mol).density_fit(auxbasis=case.pyscf_basis).to_gpu()
engine.conv_tol=1e-12;engine.conv_tol_grad=1e-10;engine.max_cycle=100;engine.direct_scf_tol=1e-14
energy=engine.kernel()
# Stock GPU4PySCF's gradient deliberately resets its DF value cache. Capture
# the prepared auxiliary topology before that normal lifetime transition.
naux=engine.with_df.auxmol.nao
gradient=engine.nuc_grad_method();gradient.auxbasis_response=True
forces=cp.asnumpy(-gradient.kernel());cp.cuda.Stream.null.synchronize()
record={'case':CASES[a.aos],'basis':case.pyscf_basis,'auxiliary_basis':case.pyscf_basis,
 'basis_representation':case.basis_representation,'coordinates_bohr':positions.tolist(),
 'energy_hartree':float(energy),'forces_hartree_per_bohr':forces.tolist(),'converged':bool(engine.converged),
 'ao_count':mol.nao_nr(),'auxiliary_count':naux,
 'slurm_job_id':os.environ['SLURM_JOB_ID'],'versions':{'gpu4pyscf':gpu4pyscf.__version__,'pyscf':pyscf.__version__,'cupy':cp.__version__},
 'imports':{m.__name__:m.__file__ for m in (gpu4pyscf,pyscf,cp)},
 'runner_sha256':hashlib.sha256(Path(__file__).read_bytes()).hexdigest()}
a.output.parent.mkdir(parents=True,exist_ok=True);a.output.write_text(json.dumps(record,indent=2)+'\n')
assert engine.converged and np.isfinite(energy) and np.isfinite(forces).all()
print(a.aos,energy,'independent changed-geometry force reference complete',flush=True)
