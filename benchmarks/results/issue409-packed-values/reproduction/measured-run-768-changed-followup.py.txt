"""Finish only the interrupted 768 changed-geometry cell after terminal Slurm state.

A complete clean series needs just separate diagnostics. Otherwise the retry
is a standalone seven-pair series; old observations are never pooled into it.
"""

import hashlib
import json
import os
import subprocess
from pathlib import Path

assert os.environ.get("SLURM_JOB_ID")
root = Path.cwd()
prior = root / ".artifacts/issue409/rebuild-v2/768-changed-clean.json"
data = json.loads(prior.read_text())
# Slurm accounting is disabled here and completed jobs may leave scontrol
# before this queued companion starts. The terminal record was captured and
# bound to the final clean file before submission of the followup.
terminal_path = prior.with_name("terminal-9845.json")
terminal = json.loads(terminal_path.read_text())
assert terminal["clean_sha256"] == hashlib.sha256(prior.read_bytes()).hexdigest()
accounting = terminal["scheduler_record"]
state = terminal["state"]
assert state in {"COMPLETED", "TIMEOUT", "FAILED", "CANCELLED"}, state
order = [
    (i, p)
    for i in range(7)
    for p in (("dense", "packed") if i % 2 == 0 else ("packed", "dense"))
]
complete = (
    not data.get("failure")
    and [(s["repeat"], s["policy"]) for s in data["samples"]] == order
)
if complete and len(data["diagnostics"]) == 2:
    raise SystemExit("predecessor already complete; no followup necessary")
name = "rebuild-768-changed-diagnostics" if complete else "rebuild-768-changed-retry"
out = root / ".artifacts/issue409" / name
out.mkdir(exist_ok=False)
env = {k: v for k, v in os.environ.items() if not k.startswith("VIBEQC_")}
env.update(
    PYTHONPATH="python:.",
    OMP_NUM_THREADS="1",
    OPENBLAS_NUM_THREADS="1",
    LD_LIBRARY_PATH="/group/software/cuda-12.9.1/lib64:"
    + env.get("LD_LIBRARY_PATH", ""),
    VIBEQC_LIBRARY=str(root / ".artifacts/issue409/stage2/frozen/libvibeqc.so"),
    VIBEQC_DF_REFERENCE_FINAL_VALIDATION="0",
    VIBEQC_DF_FINAL_PROJECTION="auto",
    VIBEQC_DF_FORCE_SCREEN_ABS="off",
    VIBEQC_DF_EXCHANGE="auto",
    VIBEQC_DF_SEED_EXCHANGE="auto",
    VIBEQC_DF_FINAL_EXCHANGE="auto",
    VIBEQC_DF_RESIDENT_EXCHANGE="auto",
    VIBEQC_DF_SHELL_POLICY="auto",
)
script = "rebuild-diagnostic-companion.py" if complete else "rebuild-endpoints.py"
command = [
    "/tmp/vibeqc-pr-review-env/bin/python",
    ".artifacts/issue409/stage2/" + script,
    "--aos",
    "768",
    "--phase",
    "changed",
    "--repeats",
    "0" if complete else "7",
    "--diagnostics-after",
    "--output",
    str(out / "768-changed-clean.json"),
    "--checkpoint",
    "/home/jzzeng/codes/vibeqc-issues388-391/.artifacts/issues388-391/final/ablations/768-VIBEQC_DF_DIIS_DOTS.checkpoint",
    "--changed-reference",
    str(root / ".artifacts/issue409/rebuild-v2/768-changed-reference.json"),
]
payload = {
    "scope": __doc__,
    "slurm_job_id": os.environ["SLURM_JOB_ID"],
    "predecessor_accounting": accounting,
    "predecessor": str(prior),
    "predecessor_sha256": hashlib.sha256(prior.read_bytes()).hexdigest(),
    "predecessor_samples": len(data["samples"]),
    "predecessor_diagnostics": len(data["diagnostics"]),
    "diagnostics_only": complete,
    "command": command,
    "status": "running",
    "runner_sha256": hashlib.sha256(Path(__file__).read_bytes()).hexdigest(),
}
manifest = out / "manifest.json"
manifest.write_text(json.dumps(payload, indent=2) + "\n")
with (out / "768-changed-clean.log").open("x") as log:
    result = subprocess.run(
        command, env=env, stdout=log, stderr=subprocess.STDOUT, check=False
    )
payload.update(
    exit_code=result.returncode, status="passed" if result.returncode == 0 else "failed"
)
manifest.write_text(json.dumps(payload, indent=2) + "\n")
print(payload["status"], "diagnostics_only", complete, flush=True)
raise SystemExit(result.returncode)
