"""Stage-2 validation, invoked only after the build and inside finite Slurm."""
import hashlib
import json
import os
from pathlib import Path
import subprocess

assert os.environ.get('SLURM_JOB_ID')
root = Path.cwd()
out = root / '.artifacts/issue409/stage2'
env = {k:v for k,v in os.environ.items() if not k.startswith('VIBEQC_')}
env.update(PYTHONPATH='python:.', OMP_NUM_THREADS='1', OPENBLAS_NUM_THREADS='1',
    LD_LIBRARY_PATH='/group/software/cuda-12.9.1/lib64:' + env.get('LD_LIBRARY_PATH', ''),
    VIBEQC_LIBRARY=str(root / 'build/cuda/libvibeqc.so'), VIBEQC_RESOURCE_CUDA_TEST='1')
commands = [
    ('density-tests', ['./build/cuda/vibeqc_density_fitting_tests']),
    ('response-tests', ['./build/cuda/vibeqc_df_occupied_response_tests']),
    ('molecular-tests', ['/tmp/vibeqc-pr-review-env/bin/python', '-m', 'pytest', '-x', '-vv',
        'tests/python/test_df_packed_values_cuda.py', '--basetemp', str(out / 'pytest-tmp')]),
]
paths = ['build/cuda/libvibeqc.so','build/cuda/vibeqc_density_fitting_tests',
    'build/cuda/vibeqc_df_occupied_response_tests','build/cuda/generated/build_identity.hpp']
record = dict(slurm_job_id=os.environ['SLURM_JOB_ID'],
    cuda_visible_devices=os.environ.get('CUDA_VISIBLE_DEVICES'),
    sha256={p:hashlib.sha256((root/p).read_bytes()).hexdigest() for p in paths}, checks=[])
for name, command in commands:
    log = out / f'{name}.log'
    run_env = env.copy()
    if name == 'response-tests':
        run_env['VIBEQC_DF_TRACE'] = str(out / 'response-trace.jsonl')
    with log.open('w') as stream:
        result = subprocess.run(command, env=run_env, stdout=stream, stderr=subprocess.STDOUT)
    record['checks'].append(dict(name=name, command=command, exit_code=result.returncode,
        log_sha256=hashlib.sha256(log.read_bytes()).hexdigest(), tail=log.read_text().splitlines()[-10:]))
    (out / 'validation.json').write_text(json.dumps(record,indent=2)+'\n')
    print(name, result.returncode, record['checks'][-1]['tail'], flush=True)
    if result.returncode:
        raise SystemExit(result.returncode)
