"""Profile frozen native warm dataflow after clean timing, without a rebuild.

Each representation runs in a fresh child process. CUPTI transfers and sampled
process memory are intrusive diagnostics, never additional clean samples.
"""

import hashlib
import json
import os
import subprocess
from pathlib import Path

assert os.environ.get("SLURM_JOB_ID")
root = Path.cwd()
out = root / ".artifacts/issue409/dataflow-profiles"
out.mkdir(exist_ok=False)
env = {key: value for key, value in os.environ.items() if not key.startswith("VIBEQC_")}
env.update(
    PYTHONPATH="python:.",
    OMP_NUM_THREADS="1",
    OPENBLAS_NUM_THREADS="1",
    LD_LIBRARY_PATH="/group/software/cuda-12.9.1/lib64:"
    + env.get("LD_LIBRARY_PATH", ""),
    VIBEQC_LIBRARY=str(root / ".artifacts/issue409/stage2/frozen/libvibeqc.so"),
    VIBEQC_DF_REFERENCE_FINAL_VALIDATION="0",
    VIBEQC_DF_FINAL_PROJECTION="auto",
    VIBEQC_DF_FORCE_SCREEN_ABS="off",
    VIBEQC_DF_EXCHANGE="auto",
    VIBEQC_DF_SEED_EXCHANGE="auto",
    VIBEQC_DF_FINAL_EXCHANGE="auto",
    VIBEQC_DF_RESIDENT_EXCHANGE="auto",
    VIBEQC_DF_SHELL_POLICY="auto",
)
nsys = "/group/software/cuda-12.9.1/nsight-systems-2025.1.3/bin/nsys"
manifest = {
    "scope": __doc__,
    "slurm_job_id": os.environ["SLURM_JOB_ID"],
    "runner_sha256": hashlib.sha256(Path(__file__).read_bytes()).hexdigest(),
    "checks": [],
}


def run(name, command):
    check = {"name": name, "command": command, "status": "running"}
    manifest["checks"].append(check)
    save()
    with (out / (name + ".log")).open("x") as log:
        result = subprocess.run(
            command, env=env, stdout=log, stderr=subprocess.STDOUT, check=False
        )
    check.update(
        exit_code=result.returncode,
        status="passed" if not result.returncode else "failed",
    )
    check["log_sha256"] = hashlib.sha256(
        (out / (name + ".log")).read_bytes()
    ).hexdigest()
    save()
    print(name, result.returncode, flush=True)
    if result.returncode:
        raise SystemExit(result.returncode)


def save():
    (out / "manifest.json").write_text(json.dumps(manifest, indent=2) + "\n")


for aos in (192, 384, 768):
    checkpoint = (
        f"/home/jzzeng/codes/vibeqc-issues388-391/.artifacts/issues388-391/final/ablations/{aos}-VIBEQC_DF_DIIS_DOTS.checkpoint"
        if aos >= 384
        else f"/home/jzzeng/codes/vibeqc-issue408/.artifacts/issue408/verified/{aos}.checkpoint"
    )
    clean = json.loads(
        (root / f".artifacts/issue409/clean-warm/{aos}-forces.json").read_text()
    )
    assert len(clean["samples"]) == 14 and not clean.get("failure")
    assert all(
        s["maximum_energy_error"] <= 1e-9 and s["maximum_force_error"] <= 1e-8
        for s in clean["samples"]
    )
    for storage in ("dense", "packed"):
        name = f"{aos}-{storage}"
        prefix = str(out / name)
        run(
            name,
            [
                nsys,
                "profile",
                "--trace=cuda,nvtx",
                "--cuda-graph-trace=node",
                "--sample=none",
                "--cpuctxsw=none",
                "--capture-range=cudaProfilerApi",
                "--capture-range-end=stop",
                "--force-overwrite=false",
                "--output",
                prefix,
                "/tmp/vibeqc-pr-review-env/bin/python",
                ".artifacts/issue409/stage2/dataflow-probe.py",
                "--aos",
                str(aos),
                "--storage",
                storage,
                "--budget",
                "0",
                "--checkpoint",
                checkpoint,
                "--output",
                prefix + ".json",
            ],
        )
        run(
            name + "-stats",
            [
                nsys,
                "stats",
                "--report",
                "cuda_api_sum,cuda_gpu_kern_sum,cuda_gpu_mem_time_sum,cuda_gpu_mem_size_sum,nvtx_sum",
                "--format=csv",
                "--output",
                prefix,
                prefix + ".nsys-rep",
            ],
        )
manifest["status"] = "passed"
save()
