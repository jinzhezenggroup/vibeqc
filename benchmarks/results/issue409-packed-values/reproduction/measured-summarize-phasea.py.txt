"""Derive Phase-A evidence from complete cells without pooling SCF branches.

Logical operator counts combine executed eager traces with source-bound graph
replay counts. They are never kernel counts or inferred component timings.
Partial files and intrusive diagnostics cannot supply clean sample medians.
"""
import collections
import hashlib
import importlib.util
import json
from pathlib import Path
import numpy as np

root=Path('.artifacts/issue409');source=Path('src/scf/cuda/df_rhf_scf.cpp')
files=sorted((root/'clean-warm').glob('*-forces.json'))+sorted((root/'clean-warm').glob('*-energy.json'))
files+=[p for p in sorted(root.glob('rebuild*/*-clean.json')) if p.parent.name != 'rebuild-768-changed-diagnostics']+sorted(root.glob('unequal*/*-measure.json'))
files+=sorted((root/'constrained-clean').glob('*.json'))
# Bind the separate diagnostic process without replacing or pooling any clean
# sample. The collector verifies binary, model, geometry and density identity.
spec=importlib.util.spec_from_file_location('packed_evidence_collector',Path('benchmarks/experiments/issue409-packed-values/collect.py'))
collector=importlib.util.module_from_spec(spec)
spec.loader.exec_module(collector)
result={'scope':__doc__,'cells':[],'incomplete':[]}
for path in files:
    data=json.loads(path.read_text())
    if path == root/'rebuild-v2/768-changed-clean.json':
        composed=collector.bind_changed_diagnostics(path,root/'rebuild-768-changed-diagnostics')
        if composed is not None:data=composed
    if 'samples' not in data:continue
    samples={p:{s['repeat']:s for s in data['samples'] if s['policy']==p} for p in ('dense','packed')}
    if data.get('failure') or any(set(v)!=set(range(7)) for v in samples.values()):
        result['incomplete'].append({'source':str(path),'failure':data.get('failure'),
            'counts':{p:len(v) for p,v in samples.items()}});continue
    times={p:np.array([v[i]['seconds'] for i in range(7)]) for p,v in samples.items()}
    ix=np.random.default_rng(20260917).integers(0,7,size=(20000,7))
    ratio=np.median(times['packed'][ix],axis=1)/np.median(times['dense'][ix],axis=1)
    medians={p:float(np.median(t)) for p,t in times.items()}
    row={'source':str(path),'sha256':hashlib.sha256(path.read_bytes()).hexdigest(),
        'phase':data.get('phase','warm'),'aos':data['aos'],'pairs':7,
        'medians_seconds':medians,'packed_over_dense':medians['packed']/medians['dense'],
        'paired_bootstrap_ratio_95':np.quantile(ratio,[.025,.975]).tolist(),
        'work_distributions':{p:dict(collections.Counter(str(v[i]['iterations']) for i in range(7))) for p,v in samples.items()},
        'same_scf_updates':all(samples['dense'][i]['iterations']==samples['packed'][i]['iterations'] for i in range(7)),
        'maximum_energy_error':max(s['maximum_energy_error'] for s in data['samples']),
        'maximum_force_error':max((s['maximum_force_error'] or 0) for s in data['samples']),
        'diagnostic_work':[]}
    if 'diagnostic_companion' in data:row['diagnostic_companion']=data['diagnostic_companion']
    assert row['maximum_energy_error']<=1e-9 and row['maximum_force_error']<=1e-8
    for sample in data.get('diagnostics',[]):
        policy=sample['policy'];paths=sample.get('traces',{})
        journal=Path(paths['VIBEQC_DF_PROGRESS_TRACE']) if paths else path.with_suffix(f'.diagnostic-0-{policy}.journal.jsonl')
        if not journal.exists():continue
        begins=collections.Counter();values=collections.defaultdict(list)
        for line in journal.read_text().splitlines():
            event=json.loads(line)
            if event['event']=='BEGIN':begins[event['name']]+=1
            if event['event']=='VALUE':values[event['name']+':'+event['key']].append(event['value'])
        iterations=sample['iterations'];iterations=iterations[0] if isinstance(iterations,list) else iterations
        assert begins['compact_rhf_scf']==1
        assert values['compact_iteration_readback:device_iterations']==list(range(1,iterations+1))
        assert set(values['compact_iteration_readback:system'])=={0}
        assert values['compact_iteration_readback:converged']==[0]*(iterations-1)+[1]
        replays=sum(values['compact_rhf_scf:host_graph_replay'])
        groups={g['operation']:g for g in sample['components']['groups'] if g['execution']=='stream'}
        eager={name:groups.get(name,{}).get('calls',0) for name in ('ri_j','ri_k','ri_k_occupied','compact_eigensolve','scf_seed_iteration')}
        assert replays<=iterations-eager['scf_seed_iteration']
        final={key:sum(values['finalization:'+key]) for key in ('final_fock_evaluations','final_eigen_solves','final_density_updates','final_candidate_rejections')}
        counts={'J':eager['ri_j']+replays,'K':eager['ri_k']+eager['ri_k_occupied']+replays,
            'eigensystems_including_density_seed':eager['compact_eigensolve']+replays}
        assert counts['J']==counts['K']==iterations+final['final_fock_evaluations']
        counters=groups.get('force_response',{}).get('counter_sums',{})
        row['diagnostic_work'].append({'policy':policy,'iterations':iterations,
            'logical_operators':counts,'eager_operations':eager,'graph_replays':replays,'finalization':final,
            'force_dataflow':{k:v for k,v in counters.items() if k.startswith(('raw_','response_ao_','response_density_','response_metric_','response_final_projection','response_occupied','response_borrowed','host_to_device','device_to_host','tensor_host'))},
            'journal':str(journal),'journal_sha256':hashlib.sha256(journal.read_bytes()).hexdigest(),
            'graph_body_source':str(source),'graph_body_source_sha256':hashlib.sha256(source.read_bytes()).hexdigest()})
    result['cells'].append(row)
(root/'phasea-summary.json').write_text(json.dumps(result,indent=2)+'\n')
for row in result['cells']:
    print(row['source'],row['medians_seconds'],'ratio',round(row['packed_over_dense'],5),
        'same_updates',row['same_scf_updates'],'diagnosed',len(row['diagnostic_work']))
