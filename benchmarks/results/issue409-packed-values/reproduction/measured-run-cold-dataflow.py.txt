"""Collect cold native dataflow for 192/384/768 and unequal 96/464 dimensions.

Each arm runs in a fresh process with the frozen library. These intrusive
profiles include preparation, integral generation and whitening; they do not
replace or supplement the clean timing repeats. No build occurs in this job.
"""

from __future__ import annotations

import hashlib
import json
import os
import subprocess
from pathlib import Path


def main():
    assert os.environ.get("SLURM_JOB_ID")
    root = Path.cwd()
    out = root / ".artifacts/issue409/cold-dataflow-profiles"
    out.mkdir(exist_ok=False)
    env = {k: v for k, v in os.environ.items() if not k.startswith("VIBEQC_")}
    env.update(
        PYTHONPATH="python:.",
        OMP_NUM_THREADS="1",
        OPENBLAS_NUM_THREADS="1",
        LD_LIBRARY_PATH="/group/software/cuda-12.9.1/lib64:"
        + env.get("LD_LIBRARY_PATH", ""),
        VIBEQC_LIBRARY=str(root / ".artifacts/issue409/stage2/frozen/libvibeqc.so"),
        VIBEQC_DF_REFERENCE_FINAL_VALIDATION="0",
        VIBEQC_DF_FINAL_PROJECTION="auto",
        VIBEQC_DF_FORCE_SCREEN_ABS="off",
        VIBEQC_DF_EXCHANGE="auto",
        VIBEQC_DF_SEED_EXCHANGE="auto",
        VIBEQC_DF_FINAL_EXCHANGE="auto",
        VIBEQC_DF_RESIDENT_EXCHANGE="auto",
        VIBEQC_DF_SHELL_POLICY="auto",
    )
    nsys = "/group/software/cuda-12.9.1/nsight-systems-2025.1.3/bin/nsys"
    unequal = (
        root
        / ".artifacts/issue409/unequal-v2/water-tetramer-def2-svp-spherical-reference.json"
    )
    points = [
        (str(n), "cold-dataflow-probe.py", ["--aos", str(n)]) for n in (192, 384, 768)
    ]
    points.append(
        ("96-464", "unequal-cold-dataflow-probe.py", ["--reference", str(unequal)])
    )
    payload = {
        "scope": __doc__,
        "slurm_job_id": os.environ["SLURM_JOB_ID"],
        "cuda_visible_devices": os.environ.get("CUDA_VISIBLE_DEVICES"),
        "runner_sha256": hashlib.sha256(Path(__file__).read_bytes()).hexdigest(),
        "library_sha256": hashlib.sha256(
            Path(env["VIBEQC_LIBRARY"]).read_bytes()
        ).hexdigest(),
        "controls": {k: v for k, v in env.items() if k.startswith("VIBEQC_")},
        "checks": [],
    }

    def save():
        (out / "manifest.json").write_text(json.dumps(payload, indent=2) + "\n")

    def run(name, command):
        row = {"name": name, "command": command, "status": "running"}
        payload["checks"].append(row)
        save()
        with (out / f"{name}.log").open("x") as log:
            result = subprocess.run(
                command, env=env, stdout=log, stderr=subprocess.STDOUT, check=False
            )
        row.update(
            exit_code=result.returncode,
            status="passed" if not result.returncode else "failed",
        )
        row["log_sha256"] = hashlib.sha256(
            (out / f"{name}.log").read_bytes()
        ).hexdigest()
        save()
        print(name, row["status"], flush=True)
        if result.returncode:
            raise SystemExit(result.returncode)

    save()
    for shape, script, arguments in points:
        for storage in ("dense", "packed"):
            name = f"{shape}-{storage}"
            prefix = str(out / name)
            run(
                name,
                [
                    nsys,
                    "profile",
                    "--trace=cuda,nvtx",
                    "--cuda-graph-trace=node",
                    "--cuda-event-trace=false",
                    "--sample=none",
                    "--cpuctxsw=none",
                    "--capture-range=cudaProfilerApi",
                    "--capture-range-end=stop",
                    "--force-overwrite=false",
                    "--output",
                    prefix,
                    "/tmp/vibeqc-pr-review-env/bin/python",
                    ".artifacts/issue409/stage2/" + script,
                    *arguments,
                    "--storage",
                    storage,
                    "--budget",
                    "0",
                    "--output",
                    prefix + ".json",
                ],
            )
            run(
                name + "-stats",
                [
                    nsys,
                    "stats",
                    "--report",
                    "cuda_api_sum,cuda_gpu_kern_sum,cuda_gpu_mem_time_sum,cuda_gpu_mem_size_sum,nvtx_sum",
                    "--format=csv",
                    "--output",
                    prefix,
                    prefix + ".nsys-rep",
                ],
            )
    payload["status"] = "passed"
    save()


if __name__ == "__main__":
    main()
