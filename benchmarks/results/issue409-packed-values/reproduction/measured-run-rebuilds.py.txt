"""Seven interleaved cold/changed pairs, preceded by a changed-path harness check.

Only child GPU processes inside this finite Slurm allocation execute models.
The smoke record is separate and is never pooled into the declared seven pairs.
"""
import hashlib
import json
import os
from pathlib import Path
import subprocess

assert os.environ.get('SLURM_JOB_ID')
root=Path.cwd();out=root/'.artifacts/issue409/rebuild-v2';out.mkdir(exist_ok=False)
env={k:v for k,v in os.environ.items() if not k.startswith('VIBEQC_')}
env.update(PYTHONPATH='python:.',OMP_NUM_THREADS='1',OPENBLAS_NUM_THREADS='1',
    LD_LIBRARY_PATH='/tmp/vibeqc-308-gpu4pyscf-env/lib/python3.13/site-packages/cutensor/lib:/group/software/cuda-12.9.1/lib64:'+env.get('LD_LIBRARY_PATH',''),
    VIBEQC_LIBRARY=str(root/'.artifacts/issue409/stage2/frozen/libvibeqc.so'),
    VIBEQC_DF_REFERENCE_FINAL_VALIDATION='0',VIBEQC_DF_FINAL_PROJECTION='auto',
    VIBEQC_DF_FORCE_SCREEN_ABS='off',VIBEQC_DF_EXCHANGE='auto',
    VIBEQC_DF_SEED_EXCHANGE='auto',VIBEQC_DF_FINAL_EXCHANGE='auto',
    VIBEQC_DF_RESIDENT_EXCHANGE='auto',VIBEQC_DF_SHELL_POLICY='auto')
checks=[]
def run(name,command):
    logfile=out/f'{name}.log'
    with logfile.open('x') as stream:
        print('starting',name,flush=True)
        result=subprocess.run(command,env=env,stdout=stream,stderr=subprocess.STDOUT)
    checks.append({'name':name,'command':command,'exit_code':result.returncode,
        'log_sha256':hashlib.sha256(logfile.read_bytes()).hexdigest()})
    (out/f'checks-{os.environ["SLURM_JOB_ID"]}.json').write_text(json.dumps({
        'slurm_job_id':os.environ['SLURM_JOB_ID'],'checks':checks,
        'runner_sha256':hashlib.sha256(Path(__file__).read_bytes()).hexdigest()},indent=2)+'\n')
    print(name,result.returncode,logfile.read_text().splitlines()[-3:],flush=True)
    if result.returncode:raise SystemExit(result.returncode)

for n in (384,768):
    oracle=out/f'{n}-changed-reference.json'
    run(f'{n}-oracle',['/tmp/vibeqc-308-gpu4pyscf-env/bin/python',
        '.artifacts/issue409/stage2/changed-reference.py','--aos',str(n),'--output',str(oracle)])
    checkpoint=f'/home/jzzeng/codes/vibeqc-issues388-391/.artifacts/issues388-391/final/ablations/{n}-VIBEQC_DF_DIIS_DOTS.checkpoint'
    for phase,repeats,suffix in ([('changed',1,'smoke')] if n==384 else [])+[
            ('cold',7,'clean'),('changed',7,'clean')]:
        name=f'{n}-{phase}-{suffix}'
        command=['/tmp/vibeqc-pr-review-env/bin/python',
            '.artifacts/issue409/stage2/rebuild-endpoints.py','--aos',str(n),
            '--phase',phase,'--output',str(out/f'{name}.json'),'--repeats',str(repeats)]
        if phase=='changed':command+=['--checkpoint',checkpoint,'--changed-reference',str(oracle)]
        if suffix=='clean':command.append('--diagnostics-after')
        run(name,command)
