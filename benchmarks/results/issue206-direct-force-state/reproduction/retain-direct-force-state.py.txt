"""Retain bounded scientific evidence; binaries and routine logs stay local."""

import hashlib
import json
from pathlib import Path
import shutil


def main():
    source = Path(".artifacts/issue206")
    target = Path("benchmarks/results/issue206-direct-force-state")
    target.mkdir(exist_ok=False)
    copies = {
        "cpu-force-diagnosis/report.json": "diagnosis/tight-cpu-oracles.json",
        "weighted-density-study-summary.json": "history/initial-pulay-study.json",
        "oh-final-contract-diagnosis/report.json": "diagnosis/oh-physical-residual.json",
        "final-96b1-pair-diagnosis.json": "diagnosis/final-96b1-pair.json",
        "final-direct-pair-diagnosis.json": "diagnosis/final-direct-pairs.json",
        "direct-weight-build/manifest.json": "history/pulay-build.json",
        "direct-weight-build/candidate.patch": "history/pulay-source.patch",
        "direct-weight-qualification-v2/baseline/report.json": "history/baseline-accuracy.json",
        "direct-weight-qualification-v2/candidate/report.json": "history/pulay-accuracy.json",
        "direct-force-polish-build/manifest.json": "history/one-update-build.json",
        "direct-force-polish-build/candidate.patch": "history/one-update-source.patch",
        "direct-force-polish-qualification/candidate/report.json": "history/one-update-incomplete-v1.json",
        "direct-force-polish-qualification-v2/candidate/report.json": "history/one-update-incomplete-v2.json",
        "direct-force-polish-qualification-v3/candidate/report.json": "history/one-update-accuracy.json",
        "direct-force-residual-build/manifest.json": "preliminary/build.json",
        "direct-force-residual-build/candidate.patch": "preliminary/measured-source.patch",
        "direct-force-residual-native-tests/summary.json": "preliminary/native-tests.json",
        "direct-force-residual-qualification/candidate/report.json": "preliminary/accuracy.json",
        "direct-force-residual-qualification/progress.jsonl": "preliminary/force-work.jsonl",
        "consistent-direct-matrix/manifest.json": "history/pulay-matrix/campaign.json",
        "direct-force-residual-matrix/manifest.json": "preliminary/matrix/campaign.json",
        "direct-force-final-build/manifest.json": "candidate/build.json",
        "direct-force-final-build/candidate.patch": "candidate/measured-source.patch",
        "direct-force-final-native-tests/summary.json": "candidate/native-tests.json",
        "direct-force-final-validation/summary.json": "candidate/validation.json",
        "direct-force-final-work-summary.json": "candidate/work-summary.json",
        "direct-force-final-qualification/candidate/report.json": "candidate/accuracy.json",
        "direct-force-final-qualification/progress.jsonl": "candidate/force-work.jsonl",
        "oh-final-qualified-state/report.json": "candidate/independent-final-state.json",
        "direct-force-final-matrix/manifest.json": "candidate/matrix/campaign.json",
        "direct-force-ownership/comparison.json": "candidate/ownership-delta.json",
    }
    for directory, output in (
        ("consistent-direct-matrix/matrix", "history/pulay-matrix"),
        ("direct-force-residual-matrix/matrix", "preliminary/matrix"),
        ("direct-force-final-matrix/matrix", "candidate/matrix"),
    ):
        for path in (source / directory).glob("*.json"):
            copies[str(path.relative_to(source))] = f"{output}/{path.name}"
    runners = (
        "diagnose-oh-final-contract.py", "diagnose-direct-weighted-density.py",
        "diagnose-frozen-direct-weight.py", "qualify-direct-force-polish.py",
        "qualify-direct-force-polish-v2.py", "qualify-direct-force-polish-v3.py",
        "qualify-direct-force-residual.py", "run-direct-force-residual.py",
        "run-native-direct-force-residual.py", "retain-direct-force-state.py",
        "validate-direct-force-final.py", "diagnose-final-direct-pairs.py", "run-native-direct-force-final.py",
        "qualify-direct-force-final.py", "diagnose-oh-qualified-state.py", "run-direct-force-final.py",
    )
    for runner in runners:
        copies[runner] = f"reproduction/{runner}.txt"
    entries = []
    for old, new in copies.items():
        original, destination = source / old, target / new
        data = original.read_bytes()
        assert len(data) < 1 << 20, old
        destination.parent.mkdir(parents=True, exist_ok=True)
        shutil.copyfile(original, destination)
        entries.append({"path": new, "source": str(original), "bytes": len(data),
                        "sha256": hashlib.sha256(data).hexdigest()})
    campaign = json.loads((source / "direct-force-final-matrix/manifest.json").read_text())
    assert campaign["status"] != "running"
    manifest = {
        "scope": __doc__, "status": "local scientific candidate, not general performance qualification",
        "original_controls_and_gates_unchanged": True,
        "candidate_matrix_status": campaign["status"], "files": entries,
        "partial_runner_failures": {
            "history/one-update-incomplete-v1.json": "Runner wrongly asserted zero total trace-file growth; generic batch records append even for energy-only calls.",
            "history/one-update-incomplete-v2.json": "Runner omitted changed coordinates from the energy-only call; version three explicitly supplied them.",
        },
        "limits": [
            "Historical failures remain failed; partial reports retain their original running field after runner assertion aborts.",
            "CPU correction trajectories are diagnosis only; production has no CPU/PySCF scientific dependency.",
            "The matrix compares ordinary converged endpoints; differing iteration branches do not establish equal operator work.",
            "#206 remains open for DF, cold/changed/energy-only, small/UHF external and resource/unequal-auxiliary coverage.",
        ],
    }
    (target / "manifest.json").write_text(json.dumps(manifest, indent=2) + "\n")
    print(len(entries), "retained indexed evidence records")


if __name__ == "__main__":
    main()
