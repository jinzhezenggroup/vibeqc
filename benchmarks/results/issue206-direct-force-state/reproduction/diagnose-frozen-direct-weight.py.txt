"""Reconstruct the unchanged one-update direct branch with independent CPU Fock.

The captured density is the immutable post-cold input, not an exported final
state. CPU F(P0) reconstructs the single accepted update P1; CPU F(P1) then
reconstructs the final canonical weighted density. Agreement with actual native
energies/full forces bounds reconstruction error. P1 F(P1) P1 / 2 is a separate
diagnostic candidate; no native code, control, timing or acceptance is changed.
"""

import hashlib
import json
import os
from pathlib import Path

import numpy as np
from pyscf import gto, scf

from benchmarks.compare_gpu4pyscf_batch import native_build_metadata
from vibeqc import Calculator
from vibeqc.progressive import _retained_density


def main():
    assert os.environ.get("SLURM_JOB_ID")
    assert not os.environ.get("VIBEQC_FINAL_FOCK_REBUILD")
    out = Path(".artifacts/issue206/frozen-direct-weight")
    out.mkdir(exist_ok=False)
    path = Path(".artifacts/issue206/aot-direct-matrix/matrix/water-tetramer-def2-svp-spherical-b4-none.json")
    data = json.loads(path.read_text())
    atoms = [[(a["element"], a["coordinates_bohr"]) for a in g]
             for g in data["workload"]["geometries"]]
    opath = Path(".artifacts/issue206/cpu-force-diagnosis/report.json")
    oracle = next(x for x in json.loads(opath.read_text())["cases"] if x["label"] == "direct-96-b4")
    assert json.loads(Path(oracle["original_path"]).read_text())["workload"]["geometries"] == data["workload"]["geometries"]
    calc = Calculator(method="rhf", basis="def2-svp", basis_representation="spherical", device="cuda",
                      density_fitting="none", max_iterations=100, energy_tolerance=1e-12,
                      density_tolerance=1e-10, screening_tolerance=1e-14)
    report = {"scope": __doc__, "slurm_job_id": os.environ["SLURM_JOB_ID"],
              "cuda_visible_devices": os.environ.get("CUDA_VISIBLE_DEVICES"),
              "native_build": native_build_metadata(calc),
              "runner_sha256": hashlib.sha256(Path(__file__).read_bytes()).hexdigest(),
              "input_sha256": hashlib.sha256(path.read_bytes()).hexdigest(),
              "oracle_sha256": hashlib.sha256(opath.read_bytes()).hexdigest(),
              "items": [], "status": "running"}

    def save():
        (out / "report.json").write_text(json.dumps(report, indent=2) + "\n")

    def maximum(x):
        return float(np.max(np.abs(x)))

    save()
    with calc.prepare_batch(atoms, charges=[0]*4, multiplicities=[1]*4, warm_start=True) as batch:
        batch.execute(strict=True)
        batch.set_warm_start_updates(False)
        seeds = [_retained_density(batch, i)[0] for i in range(4)]
        batch.execute(strict=True)
        samples = [batch.execute(strict=True) for _ in range(3)]
        assert all(item.iterations == 1 for sample in samples for item in sample.items)
        assert all(item.density_rms > 1e-12 for sample in samples for item in sample.items)
        # For density_tolerance=1e-10 the existing reuse ceiling is 1e-12:
        # these items rebuild F(P1), so the final energy/force density is P1.
        for i, seed in enumerate(seeds):
            np.testing.assert_array_equal(seed, _retained_density(batch, i)[0])
    for slot, (geometry, seed) in enumerate(zip(atoms, seeds, strict=True)):
        mol = gto.M(atom=geometry, unit="Bohr", basis="def2-svp", cart=False, verbose=0)
        mf = scf.RHF(mol)
        mf.direct_scf_tol = 1e-14
        overlap, hcore = mf.get_ovlp(), mf.get_hcore()
        occ = np.zeros(mol.nao)
        occ[:mol.nelectron//2] = 2
        _, first_coeff = mf.eig(hcore + mf.get_veff(mol, seed), overlap)
        density = (first_coeff * occ) @ first_coeff.T
        fock = hcore + mf.get_veff(mol, density)
        eps, coeff = mf.eig(fock, overlap)
        canonical = (coeff * (occ * eps)) @ coeff.T
        consistent = density @ fock @ density / 2
        energy = float(np.einsum("ij,ji", density, hcore+fock)/2 + mol.energy_nuc())
        reference = np.array(oracle["oracles"][slot]["forces"])
        mf.make_rdm1 = lambda *args, **kwargs: density
        gradient = mf.nuc_grad_method()
        forces = {}
        for name, weighted in (("canonical", canonical), ("consistent", consistent)):
            gradient.make_rdm1e = lambda *args, w=weighted, **kwargs: w
            forces[name] = -gradient.kernel(mo_energy=eps, mo_coeff=coeff, mo_occ=occ)
        rows = [{"iterations": s.items[slot].iterations, "density_rms": s.items[slot].density_rms,
                 "energy": s.items[slot].energy, "forces": s.items[slot].forces.tolist(),
                 "canonical_vs_native": maximum(forces["canonical"]-s.items[slot].forces),
                 "native_vs_oracle": maximum(s.items[slot].forces-reference),
                 "cpu_energy_error_from_native": abs(energy-s.items[slot].energy)} for s in samples]
        assert max(x["cpu_energy_error_from_native"] for x in rows) < 1e-9
        report["items"].append({"slot": slot, "native_samples": rows,
                                "canonical_reconstructed_forces": forces["canonical"].tolist(),
                                "consistent_reconstructed_forces": forces["consistent"].tolist(),
                                "cpu_oracle_forces": reference.tolist(),
                                "canonical_vs_oracle": maximum(forces["canonical"]-reference),
                                "consistent_vs_oracle": maximum(forces["consistent"]-reference),
                                "reconstructed_density_rms": float(np.sqrt(np.mean((density-seed)**2))),
                                "commutator_max": maximum(fock@density@overlap-overlap@density@fock)})
        np.save(out/f"seed-{slot}.npy", seed, allow_pickle=False)
        save()
        print(slot, max(x["canonical_vs_native"] for x in rows),
              max(x["native_vs_oracle"] for x in rows),
              report["items"][-1]["consistent_vs_oracle"], flush=True)
    report["status"] = "completed diagnostic; original acceptance remains failed"
    save()


if __name__ == "__main__":
    main()
