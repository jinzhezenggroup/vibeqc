"""Separate final-build correctness checks from subsequent clean timings."""
import hashlib
import json
import os
from pathlib import Path
import subprocess
import sys


def main():
    assert os.environ.get("SLURM_JOB_ID")
    root = Path.cwd()
    out = root / ".artifacts/issue206/direct-force-final-validation"
    out.mkdir(exist_ok=False)
    qualification = root / ".artifacts/issue206/direct-force-final-qualification"
    qualification.mkdir(exist_ok=False)
    env = {k:v for k,v in os.environ.items() if not k.startswith("VIBEQC_")}
    env.update(PYTHONPATH="python:.", OMP_NUM_THREADS="1", OPENBLAS_NUM_THREADS="1",
               VIBEQC_LIBRARY=str(root / ".artifacts/issue206/direct-force-final-build/libvibeqc.so"),
               LD_LIBRARY_PATH="/group/software/cuda-12.9.1/lib64",
               VIBEQC_RESOURCE_CUDA_TEST="1",
               VIBEQC_DF_PROGRESS_TRACE=str(qualification / "progress.jsonl"))
    cpu_python = "/tmp/vibeqc-308-gpu4pyscf-env/bin/python"
    commands = [
        [sys.executable, ".artifacts/issue206/run-native-direct-force-final.py"],
        [sys.executable, "-m", "pytest", "-q", "tests/python/test_direct_force_state_cuda.py",
         "--junitxml", str(out / "regression.xml")],
        [cpu_python, ".artifacts/issue206/qualify-direct-force-final.py", "--arm", "candidate"],
        [cpu_python, ".artifacts/issue206/diagnose-oh-qualified-state.py"],
    ]
    report = {"scope": __doc__, "slurm_job_id": os.environ["SLURM_JOB_ID"],
              "cuda_visible_devices": os.environ.get("CUDA_VISIBLE_DEVICES"),
              "runner_sha256": hashlib.sha256(Path(__file__).read_bytes()).hexdigest(),
              "status": "running", "checks": []}
    def save():
        (out / "summary.json").write_text(json.dumps(report, indent=2)+"\n")
    save()
    for index, command in enumerate(commands):
        with (out / f"check-{index}.log").open("x") as log:
            run = subprocess.run(command, env=env, stdout=log, stderr=subprocess.STDOUT)
        report["checks"].append({"command":command,"returncode":run.returncode})
        save()
        print(index, run.returncode, flush=True)
        if run.returncode:
            report["status"] = "failed"
            save()
            raise SystemExit(run.returncode)
    assert json.loads((qualification / "candidate/report.json").read_text())["status"] == "passed"
    report["status"] = "passed"
    save()


if __name__ == "__main__":
    main()
