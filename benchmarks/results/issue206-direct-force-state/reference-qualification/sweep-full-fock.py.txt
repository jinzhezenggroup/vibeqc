"""Accuracy-only stock full-Fock reference sweep; historical failures remain failures.

Run under an allocated GPU. Inputs are the retained exact-geometry paired record
and independent CPU diagnosis. Raw results go to a fresh output directory.
"""
from __future__ import annotations

import argparse
import hashlib
import json
import os
from pathlib import Path
import subprocess


def digest(path: Path) -> str:
    return hashlib.sha256(path.read_bytes()).hexdigest()


def main() -> None:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--paired-record", type=Path, required=True)
    parser.add_argument("--cpu-oracle", type=Path, required=True)
    parser.add_argument("--output-directory", type=Path, required=True)
    parser.add_argument("--repeats", type=int, default=3)
    args = parser.parse_args()
    if not os.environ.get("SLURM_JOB_ID"):
        raise RuntimeError("An allocated Slurm GPU is required")
    if args.repeats < 1:
        raise ValueError("repeats must be positive")
    import cupy as cp
    import gpu4pyscf
    import numpy as np
    import pyscf
    from gpu4pyscf.lib import cutensor
    from pyscf import gto, scf

    if cutensor.cutensor is None or cutensor.contract_engine is not None:
        raise RuntimeError("Use the original stock cuTENSOR backend")
    data = json.loads(args.paired_record.read_text())
    workload = data["workload"]
    if (workload["ao_count"], workload["batch_size"], workload["density_fitting"]) != (96, 4, "none"):
        raise ValueError("Expected the retained direct 96-AO batch-4 record")
    oracle = next(x for x in json.loads(args.cpu_oracle.read_text())["cases"]
                  if x["label"] == "direct-96-b4")
    oracle_input = json.loads(Path(oracle["original_path"]).read_text())
    if workload["geometries"] != oracle_input["workload"]["geometries"]:
        raise ValueError("CPU oracle and paired record geometries differ")
    args.output_directory.mkdir(parents=True, exist_ok=False)
    thresholds = (1e-9, 1e-10, 1e-11, 1e-12)
    margin = 3e-12
    report = {
        "scope": __doc__, "slurm_job_id": os.environ["SLURM_JOB_ID"],
        "cuda_visible_devices": os.environ.get("CUDA_VISIBLE_DEVICES"),
        "source_revision": subprocess.check_output(["git", "rev-parse", "HEAD"], text=True).strip(),
        "runner_sha256": digest(Path(__file__)),
        "paired_record_sha256": digest(args.paired_record),
        "cpu_oracle_sha256": digest(args.cpu_oracle),
        "packages": {"gpu4pyscf": gpu4pyscf.__version__, "pyscf": pyscf.__version__, "cupy": cp.__version__},
        "thresholds": thresholds, "repeats": args.repeats,
        "force_qualification_margin": margin,
        "energy_tolerance": 1e-12, "screening_tolerance": 1e-14, "max_iterations": 100,
        "seed_policy": "Independent historical 1e-9 cold density per slot, frozen for every threshold and repeat",
        "historical_gate_relabelled": False, "reference_incremental_fock": False, "status": "running", "systems": [],
    }
    def save():
        (args.output_directory / "report.json").write_text(json.dumps(report, indent=2) + "\n")

    for slot, geometry in enumerate(workload["geometries"]):
        mol = gto.M(atom=[(a["element"], a["coordinates_bohr"]) for a in geometry],
                    unit="Bohr", basis="def2-svp", cart=False, verbose=0)
        def engine(tolerance):
            mf = scf.RHF(mol).to_gpu()
            mf.conv_tol, mf.conv_tol_grad = 1e-12, tolerance
            mf.max_cycle, mf.direct_scf_tol = 100, 1e-14
            mf.direct_scf = False
            return mf
        cold = engine(thresholds[0])
        cold.direct_scf = True  # keep the historical cold-seed policy
        cold.kernel()
        seed = cold.make_rdm1().copy()
        reference = oracle["oracles"][slot]
        system = {"slot": slot, "geometry": geometry,
                  "cpu_oracle": reference,
                  "cold_converged": bool(cold.converged),
                  "frozen_density": cp.asnumpy(seed).tolist(), "settings": []}
        report["systems"].append(system)
        save()
        for tolerance in thresholds:
            mf = engine(tolerance)
            row = {"gradient_tolerance": tolerance, "samples": []}
            system["settings"].append(row)
            for repeat in range(args.repeats):
                energy = float(mf.kernel(dm0=seed.copy()))
                forces = cp.asnumpy(-mf.nuc_grad_method().kernel())
                physical_fock = mf.get_fock(dm=mf.make_rdm1())
                residual = float(cp.linalg.norm(mf.get_grad(mf.mo_coeff, mf.mo_occ, physical_fock)))
                row["samples"].append({
                    "repeat": repeat, "converged": bool(mf.converged),
                    "orbital_gradient_norm": residual, "energy": energy,
                    "forces": forces.tolist(),
                    "energy_error_vs_cpu": abs(energy-reference["energy"]),
                    "force_error_vs_cpu": float(np.max(np.abs(forces-np.asarray(reference["forces"])))),
                })
                save()
            print(slot, tolerance, "force_error", max(x["force_error_vs_cpu"] for x in row["samples"]),
                  "residual", max(x["orbital_gradient_norm"] for x in row["samples"]),
                  "converged", all(x["converged"] for x in row["samples"]), flush=True)
    summary = []
    for i, tolerance in enumerate(thresholds[:-1]):
        current = [s for system in report["systems"] for s in system["settings"][i]["samples"]]
        tighter = [s for system in report["systems"] for s in system["settings"][i+1]["samples"]]
        drift = max(float(np.max(np.abs(np.asarray(a["forces"])-np.asarray(b["forces"]))))
                    for system in report["systems"]
                    for a in system["settings"][i]["samples"]
                    for b in system["settings"][i+1]["samples"])
        qualified = (all(s["converged"] for s in current+tighter)
                     and max(s["force_error_vs_cpu"] for s in current+tighter) <= margin
                     and drift <= margin)
        summary.append({"gradient_tolerance": tolerance, "tighter_control": thresholds[i+1],
                        "maximum_force_error_vs_cpu": max(s["force_error_vs_cpu"] for s in current),
                        "maximum_force_drift_on_tightening": drift, "qualified": qualified})
    report["summary"] = summary
    candidates = [x["gradient_tolerance"] for x in summary if x["qualified"] and x["gradient_tolerance"] < thresholds[0]]
    report["qualified_gradient_tolerance"] = candidates[0] if candidates else None
    report["status"] = "qualified" if candidates else "not_qualified"
    save()
    print(json.dumps(summary, indent=2), flush=True)
    if not candidates:
        raise SystemExit(2)


if __name__ == "__main__":
    main()
