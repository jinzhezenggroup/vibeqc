"""Account the intrusive stock profile without treating wrappers as production work."""

import collections
import hashlib
import json
import sqlite3
from pathlib import Path


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


directory = Path(".artifacts/issue206/stock-diagnostic-384")
record = directory / "384-stock.json"
database = directory / "384-stock.sqlite"
data = json.loads(record.read_text())
assert not data.get("failure")
assert data["same_scf_iterations_as_clean"]
db = sqlite3.connect(f"file:{database}?mode=ro", uri=True)
db.row_factory = sqlite3.Row
copies = [
    dict(row)
    for row in db.execute(
        "select e.label as direction,count(*) as calls,sum(m.bytes) as bytes "
        "from CUPTI_ACTIVITY_KIND_MEMCPY m join ENUM_CUDA_MEMCPY_OPER e "
        "on e.id=m.copyKind group by m.copyKind"
    )
]
apis = [
    dict(row)
    for row in db.execute(
        "select s.value as name,count(*) as calls,sum(m.end-m.start) as summed_ns "
        "from CUPTI_ACTIVITY_KIND_RUNTIME m join StringIds s on s.id=m.nameId "
        "group by m.nameId order by calls desc"
    )
]
groups = collections.defaultdict(list)
for row in data["operations"]:
    groups[(row["thread"], row["operation"])].append(row)
account = {
    "scope": __doc__,
    "runner_sha256": sha(Path(__file__)),
    "source_record_sha256": sha(record),
    "source_sqlite_sha256": sha(database),
    "logical_calls": dict(
        collections.Counter(r["operation"] for r in data["operations"])
    ),
    "thread_local_scopes": [
        {
            "thread": thread,
            "operation": name,
            "calls": len(rows),
            "inclusive_ns": sum(r["inclusive_ns"] for r in rows),
            "exclusive_ns": sum(r["exclusive_ns"] for r in rows),
        }
        for (thread, name), rows in groups.items()
    ],
    "scope_limit": "Exclusive scopes subtract synchronous children in the same thread. Threads may overlap; neither their sums nor API/kernel totals form clean endpoint time.",
    "factor_observations": data["factor_observations"],
    "cuda_transfers": copies,
    "runtime_apis": apis,
    "kernel_launches": db.execute(
        "select count(*) from CUPTI_ACTIVITY_KIND_KERNEL"
    ).fetchone()[0],
    "known_wrapper_device_synchronizations": 2 * len(data["operations"]),
    "observed_device_synchronizations": sum(
        r["calls"] for r in apis if r["name"].startswith("cudaDeviceSynchronize")
    ),
    "synchronization_limit": "Wrappers deliberately fence each call twice; observed profile counts are not clean production synchronization counts.",
    "memory_limit": "These process samples include Nsight and component-wrapper overhead, so they are diagnostic observations, not unprofiled memory acceptance.",
    "sampled_device_peak": data["sampled_device_peak"],
    "sampled_host_peak": data["sampled_host_peak"],
    "pool_before": data["pool_before"],
    "pool_after": data["pool_after"],
    "numerical_parity": {
        k: data[k]
        for k in (
            "energy_error_from_clean_stock",
            "force_error_from_clean_stock",
            "same_scf_iterations_as_clean",
        )
    },
}
(directory / "account.json").write_text(json.dumps(account, indent=2) + "\n")
print(
    "Stock logical J/K:",
    account["logical_calls"]["logical_jk"],
    "eigensystems:",
    account["logical_calls"]["eigensystem"],
)
print("CUPTI transfers:", copies)
print(
    "Observed/wrapper device fences:",
    account["observed_device_synchronizations"],
    account["known_wrapper_device_synchronizations"],
)
