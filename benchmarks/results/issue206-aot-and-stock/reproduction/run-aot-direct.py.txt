"""Retain the unchanged direct matrix on an explicitly AOT-enabled release build.

This clean endpoint campaign must start only after the local build has finished.
Every original control and error gate remains authoritative. A failed gate is
retained even if the original AOT-disabled execution failure is resolved.
"""

import hashlib
import json
import os
from pathlib import Path
import shutil
import subprocess
import sys


def digest(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()


def main():
    assert os.environ.get('SLURM_JOB_ID')
    root = Path.cwd()
    source = root / 'build/cuda/libvibeqc.so'
    build = root / '.artifacts/issue206/direct-aot'
    cache = (root / 'build/cuda/CMakeCache.txt').read_text()
    assert 'VIBEQC_ENABLE_AOT_SHELLS:BOOL=ON' in cache
    assert 'CMAKE_BUILD_TYPE:STRING=Release' in cache
    assert 'VIBEQC_CUDA_FAST_COMPILE:BOOL=OFF' in cache
    assert 'Exit status: 0' in (build / 'build.log').read_text()
    out = root / '.artifacts/issue206/aot-direct-matrix'
    out.mkdir(exist_ok=False)
    library = out / 'libvibeqc.so'
    shutil.copyfile(source.resolve(), library)
    env = {k: v for k, v in os.environ.items() if not k.startswith('VIBEQC_') and k != 'CONTRACT_ENGINE'}
    env.update(PYTHONPATH='python:.', OMP_NUM_THREADS='1', OPENBLAS_NUM_THREADS='1',
               VIBEQC_LIBRARY=str(library),
               LD_LIBRARY_PATH='/tmp/vibeqc-308-gpu4pyscf-env/lib/python3.13/site-packages/cutensor/lib:'
               '/group/software/cuda-12.9.1/lib64:' + env.get('LD_LIBRARY_PATH', ''))
    command = [sys.executable, 'benchmarks/real_molecule_gate.py', '--density-fitting', 'none',
               '--repeats', '7', '--output-directory', str(out / 'matrix')]
    payload = {'scope': __doc__, 'slurm_job_id': os.environ['SLURM_JOB_ID'],
               'cuda_visible_devices': os.environ.get('CUDA_VISIBLE_DEVICES'),
               'command': command, 'status': 'running', 'points': [],
               'library_sha256': digest(library), 'cache_sha256': digest(root / 'build/cuda/CMakeCache.txt'),
               'runner_sha256': digest(__file__), 'build_log_sha256': digest(build / 'build.log'),
               'revision': subprocess.check_output(['git', 'rev-parse', 'HEAD'], text=True).strip(),
               'comparison_source_sha256': digest(root / 'benchmarks/compare_gpu4pyscf_batch.py'),
               'matrix_source_sha256': digest(root / 'benchmarks/real_molecule_gate.py')}

    def save():
        (out / 'manifest.json').write_text(json.dumps(payload, indent=2) + '\n')

    save()
    with (out / 'execution.log').open('x') as log:
        result = subprocess.run(command, env=env, stdout=log, stderr=subprocess.STDOUT, check=False)
    payload['matrix_exit_code'] = result.returncode
    summary = json.loads((out / 'matrix/summary.json').read_text())
    for point in summary['points']:
        row = {'point': point, 'all_seven_pairs_passed': False}
        path = Path(point['artifact'])
        if path.exists():
            record = json.loads(path.read_text())
            assert record['native_build']['library_sha256'] == payload['library_sha256']
            assert record['native_build']['probe']['device']['official_profile'] == 'sm_120'
            pairs = record['accuracy']['paired_warm_repeats']
            row.update(artifact_sha256=digest(path), native_build=record['native_build'],
                       paired_accuracy=pairs, timing_summary=record['timing_summary'])
            row['all_seven_pairs_passed'] = len(pairs) == 7 and all(
                x['maximum_energy_error_hartree'] <= point['maximum_energy_error_hartree']
                and x['maximum_force_error_hartree_per_bohr'] <= point['maximum_force_error_hartree_per_bohr']
                for x in pairs)
        row['passed'] = point['passed'] and row['all_seven_pairs_passed']
        payload['points'].append(row)
        save()
    assert digest(library) == payload['library_sha256']
    payload['status'] = 'passed' if result.returncode == 0 and all(x['passed'] for x in payload['points']) else 'failed'
    save()
    print(payload['status'], flush=True)
    if payload['status'] != 'passed':
        raise SystemExit(2)


if __name__ == '__main__':
    main()
