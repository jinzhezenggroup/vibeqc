"""Prepare a bounded evidence slice without modifying measured artifacts."""

import hashlib
import json
from pathlib import Path
import re
import shutil

ROOT = Path.cwd()
RUN = ROOT / ".artifacts/issue206"
OUT = RUN / "retained-aot-stock-v3"


def digest(path):
    with path.open("rb") as stream:
        return hashlib.file_digest(stream, "sha256").hexdigest()


def main():
    OUT.mkdir(exist_ok=False)
    files = []

    def retain(source, relative):
        destination = OUT / relative
        destination.parent.mkdir(parents=True, exist_ok=True)
        shutil.copyfile(source, destination)
        assert destination.stat().st_size < 1 << 20
        files.append({"path": relative, "bytes": destination.stat().st_size,
                      "sha256": digest(destination), "original_path": str(source),
                      "original_sha256": digest(source)})

    def record(relative, data):
        p = OUT / relative
        p.parent.mkdir(parents=True, exist_ok=True)
        p.write_text(json.dumps(data, indent=2, allow_nan=False) + "\n")
        files.append({"path": relative, "bytes": p.stat().st_size, "sha256": digest(p)})

    campaign = json.loads((RUN / "aot-direct-matrix/manifest.json").read_text())
    assert campaign["status"] == "failed" and campaign["matrix_exit_code"] == 2
    assert digest(ROOT / "build/cuda/CMakeCache.txt") == campaign["cache_sha256"]
    assert digest(RUN / "direct-aot/build.log") == campaign["build_log_sha256"]
    assert digest(RUN / "aot-direct-matrix/libvibeqc.so") == campaign["library_sha256"]
    retain(RUN / "aot-direct-matrix/manifest.json", "direct/campaign.json")
    retain(RUN / "aot-direct-matrix/matrix/summary.json", "direct/original-gate-summary.json")
    direct_rows = []
    oracle_path = RUN / "cpu-force-diagnosis/report.json"
    cpu_oracles = json.loads(oracle_path.read_text())
    for row in campaign["points"]:
        point = row["point"]
        path = Path(point["artifact"])
        assert digest(path) == row["artifact_sha256"]
        data = json.loads(path.read_text())
        retain(path, "direct/" + path.name)
        progress = Path(point["progress_artifact"])
        retain(progress, "direct/" + progress.name)
        pairs = data["accuracy"]["paired_warm_repeats"]
        assert len(pairs) == 7
        maximum_energy = max(x["maximum_energy_error_hartree"] for x in pairs)
        maximum_force = max(x["maximum_force_error_hartree_per_bohr"] for x in pairs)
        passed = maximum_energy <= point["maximum_energy_error_hartree"] and maximum_force <= point["maximum_force_error_hartree_per_bohr"]
        assert passed == row["all_seven_pairs_passed"]
        result = {"ao": point["ao_count"], "batch": point["batch_size"],
                  "maximum_energy_error": maximum_energy, "maximum_force_error": maximum_force,
                  "energy_gate": point["maximum_energy_error_hartree"],
                  "force_gate": point["maximum_force_error_hartree_per_bohr"],
                  "all_seven_pairs_passed": passed,
                  "ordinary_latency": data["timing_summary"]["ordinary"],
                  "iterations": {engine: [[x["iterations"] for x in s["convergence"]] for s in data[engine]["warm_samples"]]
                                 for engine in ("vibeqc", "gpu4pyscf")}}
        if point["ao_count"] == 96:
            oracle = next(x for x in cpu_oracles["cases"] if x["label"] == f'direct-96-b{point["batch_size"]}')
            old = json.loads(Path(oracle["original_path"]).read_text())
            assert old["workload"]["geometries"] == data["workload"]["geometries"]
            result["cpu_oracle_sha256"] = digest(oracle_path)
            result["maximum_force_error_vs_cpu"] = {
                engine: max(abs(component - reference)
                            for sample in data[engine]["warm_samples"]
                            for atom_forces, reference_atom_forces in zip(sample["forces_hartree_per_bohr"],
                                                                          [x["forces"] for x in oracle["oracles"]], strict=True)
                            for atom, reference_atom in zip(atom_forces, reference_atom_forces, strict=True)
                            for component, reference in zip(atom, reference_atom, strict=True))
                for engine in ("vibeqc", "gpu4pyscf")}
        direct_rows.append(result)

    cache = {}
    for line in (ROOT / "build/cuda/CMakeCache.txt").read_text().splitlines():
        if line.startswith(("CMAKE_", "VIBEQC_", "CUDAToolkit_")) and "=" in line:
            name, value = line.split("=", 1)
            if not name.endswith(":INTERNAL"):
                cache[name] = value
    log = (RUN / "direct-aot/build.log").read_text()
    record("build/configuration.json", {
        "source_identity": campaign["points"][0]["native_build"]["probe"]["source_identity"],
        "library_sha256": campaign["library_sha256"],
        "library_bytes": (RUN / "aot-direct-matrix/libvibeqc.so").stat().st_size,
        "cmake_cache_sha256": campaign["cache_sha256"], "cmake_options": cache,
        "build_command": ["cmake", "--build", "build/cuda", "-j", "6"],
        "build_log_sha256": digest(RUN / "direct-aot/build.log"),
        "build_exit_status": int(re.search(r"Exit status: (\d+)", log).group(1)),
        "wall_seconds": 33*60+29.46, "maximum_rss_kib": 3566456,
        "limit": "Observed build resources only; not a controlled clean-build comparison."})
    tests = json.loads((RUN / "aot-native-tests/summary.json").read_text())
    assert tests["status"] == "passed" and len(tests["tests"]) == 48
    assert all(not x["failures"] for x in tests["tests"])
    assert [x["name"] for x in tests["tests"] if x["skipped"]] == ["vibeqc_mp2_cuda_status_tests"]
    assert tests["library_sha256"] == campaign["library_sha256"]
    retain(RUN / "aot-native-tests/summary.json", "validation/native-tests.json")
    import xml.etree.ElementTree as ET
    optin_xml = RUN / "aot-native-tests/mp2-optin.xml"
    optin = list(ET.parse(optin_xml).iter("testcase"))
    assert len(optin) == 1 and optin[0].find("failure") is None and optin[0].find("skipped") is None
    optin_log = RUN / "aot-native-tests/mp2-optin.log"
    record("validation/mp2-optin.json", {
        "scope": "Explicitly run the CUDA allocation-status test skipped by default.",
        "slurm_job_id": re.search(r"Slurm job (\d+)", optin_log.read_text()).group(1),
        "cuda_visible_devices": re.search(r"CUDA_VISIBLE_DEVICES=(.*)", optin_log.read_text()).group(1),
        "command": ["env", "VIBEQC_MP2_CUDA_TEST=1", "ctest", "--test-dir", "build/cuda", "-R", "^vibeqc_mp2_cuda_status_tests$", "--output-on-failure"],
        "library_sha256": campaign["library_sha256"],
        "status": "passed", "test": optin[0].attrib,
        "xml_sha256": digest(optin_xml), "log_sha256": digest(optin_log)})

    stock_rows = []
    for ao in (384, 768):
        source = RUN / f"stock-diagnostic-{ao}"
        manifest = json.loads((source / "manifest.json").read_text())
        account = json.loads((source / "account.json").read_text())
        assert manifest["status"] == "passed" and manifest["stats_exit_code"] == 0
        assert digest(source / f"{ao}-stock.json") == account["source_record_sha256"]
        for p in source.iterdir():
            if p.suffix in (".json", ".csv"):
                if p.name in manifest["artifact_sha256"]:
                    assert digest(p) == manifest["artifact_sha256"][p.name]
                retain(p, f"stock-{ao}/" + p.name)
        stock_rows.append({"ao": ao, **{k:account[k] for k in (
            "logical_calls", "cuda_transfers", "known_wrapper_device_synchronizations",
            "observed_device_synchronizations", "numerical_parity", "memory_limit")},
            "branch_observation": account.get("branch_observation")})
        for name in (f"run-stock-{ao}.py", f"account-stock-{ao}.py"):
            retain(RUN / name, "reproduction/" + name + ".txt")
    for name in ("run-aot-direct.py", "run-native-aot.py", "retain-aot-stock.py"):
        retain(RUN / name, "reproduction/" + name + ".txt")
    retained = ROOT / "benchmarks/results/issue206-direct-diagnosis/reproduction"
    for name, key in (("compare_gpu4pyscf_batch.py", "comparison_source_sha256"),
                      ("real_molecule_gate.py", "matrix_source_sha256")):
        p = retained / (name + ".txt")
        assert digest(p) == campaign[key]
        retain(p, "reproduction/" + name + ".txt")
    retain(Path("/home/jzzeng/codes/vibeqc-issue409-packed/.artifacts/issue206/stock-warm-diagnostics.py"),
           "reproduction/stock-warm-diagnostics.py.txt")
    record("summary.json", {"scope": "AOT direct capability confirmation and separate intrusive stock DF profiles.",
                           "direct_matrix_passed": False, "direct": direct_rows, "stock_profiles": stock_rows,
                           "limits": ["Unchanged 96-AO strict direct force gates still fail.",
                                      "Ordinary converged latency only; no fixed-work attribution.",
                                      "No new DF speedup, unprofiled memory acceptance or DRAM-traffic measurement.",
                                      "Old failures remain preserved; #206 remains open."]})
    (OUT / "manifest.json").write_text(json.dumps({"files": files}, indent=2) + "\n")
    print(f"Prepared {len(files)} records; {sum(x['bytes'] for x in files)} bytes")


if __name__ == "__main__":
    main()
