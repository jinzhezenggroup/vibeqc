"""Validate isolated cold/changed comparison on the fresh passing 192-AO cell.

This is a clean timing job: do not build concurrently. The comparator validates
all seven original warm pairs before admitting a new endpoint campaign. A
missing or failed predecessor remains a failure and is never replaced here.
"""

from __future__ import annotations

import hashlib
import json
import os
import subprocess
from pathlib import Path


def main():
    assert os.environ.get("SLURM_JOB_ID")
    root = Path.cwd()
    out = root / ".artifacts/issue206/rebuild-smoke"
    out.mkdir(exist_ok=False)
    env = {
        k: v
        for k, v in os.environ.items()
        if not k.startswith("VIBEQC_") and k != "CONTRACT_ENGINE"
    }
    env.update(
        PYTHONPATH="python:.",
        OMP_NUM_THREADS="1",
        OPENBLAS_NUM_THREADS="1",
        LD_LIBRARY_PATH="/tmp/vibeqc-308-gpu4pyscf-env/lib/python3.13/site-packages/cutensor/lib:"
        "/group/software/cuda-12.9.1/lib64:" + env.get("LD_LIBRARY_PATH", ""),
        VIBEQC_LIBRARY=str(root / ".artifacts/issue409/stage2/frozen/libvibeqc.so"),
        VIBEQC_DF_VALUE_STORAGE="dense",
        VIBEQC_DF_REFERENCE_FINAL_VALIDATION="0",
        VIBEQC_DF_FINAL_PROJECTION="auto",
        VIBEQC_DF_FORCE_SCREEN_ABS="off",
        VIBEQC_DF_EXCHANGE="auto",
        VIBEQC_DF_SEED_EXCHANGE="auto",
        VIBEQC_DF_FINAL_EXCHANGE="auto",
        VIBEQC_DF_RESIDENT_EXCHANGE="auto",
        VIBEQC_DF_SHELL_POLICY="auto",
    )
    reference = (
        root / ".artifacts/issue206/fresh-warm-remaining/"
        "water-octamer-s4-def2-svp-spherical-b1-forces.json"
    )
    payload = {
        "scope": __doc__,
        "slurm_job_id": os.environ["SLURM_JOB_ID"],
        "runner_sha256": hashlib.sha256(Path(__file__).read_bytes()).hexdigest(),
        "checks": [],
    }
    for phase in ("cold", "changed"):
        command = [
            "/tmp/vibeqc-308-gpu4pyscf-env/bin/python",
            ".artifacts/issue206/compare-rebuilds.py",
            "--reference",
            str(reference),
            "--output",
            str(out / phase),
            "--phase",
            phase,
        ]
        row = {"phase": phase, "command": command, "status": "running"}
        payload["checks"].append(row)
        (out / "manifest.json").write_text(json.dumps(payload, indent=2) + "\n")
        with (out / f"{phase}.log").open("x") as stream:
            result = subprocess.run(
                command, env=env, stdout=stream, stderr=subprocess.STDOUT, check=False
            )
        row.update(
            exit_code=result.returncode,
            status="passed" if not result.returncode else "failed",
        )
        (out / "manifest.json").write_text(json.dumps(payload, indent=2) + "\n")
        print(phase, row["status"], flush=True)
        if result.returncode:
            raise SystemExit(result.returncode)


if __name__ == "__main__":
    main()
