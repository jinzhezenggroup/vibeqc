"""Diagnose the final direct cold failure in separate processes, without timing claims."""

import argparse
import ctypes
import hashlib
import json
import os
import subprocess
import sys
from pathlib import Path


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--library", type=Path)
    args = parser.parse_args()
    assert os.environ.get("SLURM_JOB_ID")
    out = Path(".artifacts/issue206/diagnose-direct-192")
    if args.library is None:
        out.mkdir(exist_ok=False)
        manifest = {
            "scope": __doc__,
            "slurm_job_id": os.environ["SLURM_JOB_ID"],
            "checks": [],
        }
        for label, library in (
            ("final", Path("build/cuda/libvibeqc.so")),
            ("frozen", Path(".artifacts/issue409/stage2/frozen/libvibeqc.so")),
        ):
            env = {k: v for k, v in os.environ.items() if not k.startswith("VIBEQC_")}
            env.update(
                PYTHONPATH="python:.",
                OMP_NUM_THREADS="1",
                OPENBLAS_NUM_THREADS="1",
                VIBEQC_LIBRARY=str(library.resolve()),
                CUDA_LAUNCH_BLOCKING="1",
                LD_LIBRARY_PATH="/group/software/cuda-12.9.1/lib64:"
                + env.get("LD_LIBRARY_PATH", ""),
            )
            command = [sys.executable, __file__, "--library", str(library.resolve())]
            with (out / f"{label}.log").open("x") as log:
                result = subprocess.run(
                    command, env=env, stdout=log, stderr=subprocess.STDOUT, check=False
                )
            manifest["checks"].append(
                {
                    "label": label,
                    "command": command,
                    "exit_code": result.returncode,
                    "library_sha256": hashlib.sha256(library.read_bytes()).hexdigest(),
                }
            )
            (out / "manifest.json").write_text(json.dumps(manifest, indent=2) + "\n")
        return
    from benchmarks._cases import benchmark_cases
    from vibeqc import Calculator

    case = benchmark_cases()["water-octamer-s4-def2-svp-spherical"]
    calc = Calculator(
        method=case.method,
        basis=case.vibeqc_basis,
        basis_representation=case.basis_representation,
        device="cuda",
        density_fitting="none",
        max_iterations=100,
        energy_tolerance=1e-12,
        density_tolerance=1e-10,
        screening_tolerance=1e-14,
    )
    lib = calc._library
    lib.vibeqc_get_source_identity.restype = ctypes.c_char_p
    payload = {
        "slurm_job_id": os.environ["SLURM_JOB_ID"],
        "cuda_visible_devices": os.environ.get("CUDA_VISIBLE_DEVICES"),
        "native_identity": lib.vibeqc_get_source_identity().decode(),
        "library_sha256": hashlib.sha256(args.library.read_bytes()).hexdigest(),
        "controls": {"CUDA_LAUNCH_BLOCKING": os.environ["CUDA_LAUNCH_BLOCKING"]},
    }
    with calc.prepare_batch(
        [case.atoms],
        charges=[case.charge],
        multiplicities=[case.multiplicity],
        warm_start=True,
    ) as batch:
        result = batch.execute(strict=False, properties=["energy", "forces"])
        detail = lib.vibeqc_context_get_last_detail(batch._context)
        payload["context_detail"] = detail.decode() if detail else None
        payload["items"] = [
            {
                key: getattr(item, key)
                for key in (
                    "status",
                    "status_message",
                    "converged",
                    "iterations",
                    "energy",
                )
            }
            for item in result.items
        ]
    print(json.dumps(payload, indent=2), flush=True)


if __name__ == "__main__":
    main()
