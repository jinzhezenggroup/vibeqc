"""Audit stock GPU array copy metadata only inside an allocated GPU job."""
import os
assert os.environ.get('SLURM_JOB_ID')
import hashlib, json
from pathlib import Path
import cupy as cp
from gpu4pyscf.scf.hf import make_rdm1
coeff=cp.eye(4,dtype=cp.float64);occ=cp.asarray([2.,2.,0.,0.])
tagged=make_rdm1(coeff,occ)
arrays={'make_rdm1':tagged,'snapshot_copy':tagged.copy(),'sample_copy':tagged.copy().copy(),'bare_density':cp.asarray(tagged)}
rows={name:{'type':type(a).__name__,'attributes':sorted(getattr(a,'__dict__',{})),
    'density_sha256':hashlib.sha256(cp.asnumpy(a).tobytes()).hexdigest()} for name,a in arrays.items()}
cp.cuda.Stream.null.synchronize()
result={'slurm_job_id':os.environ['SLURM_JOB_ID'],'scope':'starting information only, not a latency benchmark','arrays':rows}
Path('.artifacts/issue206/starting-information-audit.json').write_text(json.dumps(result,indent=2)+'\n')
print(json.dumps(result,indent=2))
