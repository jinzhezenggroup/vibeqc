"""Observe stock DF warm work in a separate intrusive process under Slurm.

Wrappers preserve function arguments, results and normal CDERI release. They
synchronize at scope boundaries and therefore cannot supply clean timings.
Per-thread exclusive wall scopes subtract only synchronous child scopes;
different threads may overlap and must not be added into an endpoint total.
Nsight captures the same range for actual transfers, kernels and API activity.
"""

from __future__ import annotations

import argparse
import functools
import hashlib
import importlib.metadata
import inspect
import itertools
import json
import os
import subprocess
import threading
import time
from pathlib import Path


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--case", required=True)
    parser.add_argument("--batch", type=int, required=True)
    parser.add_argument("--reference", type=Path, required=True)
    parser.add_argument("--output", type=Path, required=True)
    args = parser.parse_args()
    if not os.environ.get("SLURM_JOB_ID") or args.batch < 1:
        parser.error("finite Slurm and positive batch size required")
    if args.output.exists():
        parser.error("refusing to overwrite evidence")
    args.output.parent.mkdir(parents=True, exist_ok=True)

    import cupy as cp
    import numpy as np
    from gpu4pyscf.df import df as stock_df
    from gpu4pyscf.df import df_jk, int3c2e_bdiv
    from gpu4pyscf.df.grad import rhf as df_grad
    from gpu4pyscf.grad import rhf as hf_grad
    from gpu4pyscf.lib import cutensor
    from gpu4pyscf.scf import diis, hf, uhf
    from pyscf import gto, scf

    from benchmarks._cases import benchmark_cases
    from benchmarks.compare_gpu4pyscf_batch import _gpu_sample, scaled_geometries

    assert cutensor.cutensor is not None and cutensor.contract_engine is None
    reference = json.loads(args.reference.read_text())
    case = benchmark_cases()[args.case]
    workload = reference["workload"]
    gates = reference["settings"]["gates"]
    assert workload["case"] == args.case and workload["batch_size"] == args.batch
    assert workload["density_fitting"] == "cuda" and workload["properties"] == [
        "energy",
        "forces",
    ]
    assert reference["gate"]["passed"]
    assert len(reference["accuracy"]["paired_warm_repeats"]) >= 7
    for paired in reference["accuracy"]["paired_warm_repeats"]:
        assert (
            paired["maximum_energy_error_hartree"]
            <= gates["maximum_energy_error_hartree"]
        )
        assert (
            paired["maximum_force_error_hartree_per_bohr"]
            <= gates["maximum_force_error_hartree_per_bohr"]
        )
    # The fresh clean comparator's declared gates remain authoritative.
    rows, installed, factor_observations = [], [], []
    state = threading.local()
    identifiers = itertools.count()
    enabled = False

    def metadata(value):
        """Describe arrays without copying or retaining their physical owners."""
        if isinstance(value, (cp.ndarray, np.ndarray)):
            return {
                "type": type(value).__name__,
                "shape": list(value.shape),
                "strides": list(value.strides),
                "dtype": str(value.dtype),
                "bytes": value.nbytes,
                "tags": [
                    name
                    for name in ("mo_coeff", "mo_occ", "occ_coeff")
                    if hasattr(value, name)
                ],
            }
        if isinstance(value, (tuple, list)):
            return [metadata(item) for item in value]
        if isinstance(value, (bool, int, float, str)) or value is None:
            return value
        return {"type": type(value).__name__}

    def wrapped(function, label, *, evaluator=False, factor_owner=False):
        @functools.wraps(function)
        def call(*positional, **keywords):
            if not enabled:
                return function(*positional, **keywords)
            stack = getattr(state, "stack", None)
            if stack is None:
                stack = state.stack = []
            cp.cuda.runtime.deviceSynchronize()
            row = {
                "id": next(identifiers),
                "parent": stack[-1][0]["id"] if stack else None,
                "thread": threading.get_ident(),
                "operation": label,
                "arguments": metadata(positional),
                "keywords": {key: metadata(value) for key, value in keywords.items()},
            }
            rows.append(row)
            frame = [row, 0]
            stack.append(frame)
            cp.cuda.nvtx.RangePush("stock/" + label)
            start = time.perf_counter_ns()
            try:
                result = function(*positional, **keywords)
                row["result"] = metadata(result)
                if factor_owner:
                    owner = positional[0]
                    blocks = owner._cderi
                    values = blocks.values() if isinstance(blocks, dict) else blocks
                    factor_observations.append(
                        {
                            "operation_id": row["id"],
                            "nbf": owner.nao,
                            "naux": owner.naux,
                            "blocks": None
                            if values is None
                            else [metadata(v) for v in values],
                        }
                    )
                if evaluator:
                    # The first closure evaluates raw three-center blocks. Wrap
                    # its calls, not just construction of the closure itself.
                    result = (wrapped(result[0], "raw_three_center_block"), *result[1:])
                return result
            except Exception as error:
                row["failure"] = repr(error)
                raise
            finally:
                cp.cuda.runtime.deviceSynchronize()
                elapsed = time.perf_counter_ns() - start
                cp.cuda.nvtx.RangePop()
                row.update(inclusive_ns=elapsed, exclusive_ns=elapsed - frame[1])
                assert stack.pop() is frame
                if stack:
                    stack[-1][1] += elapsed

        return call

    def install(owner, name, label, **options):
        function = getattr(owner, name)
        source = Path(inspect.getsourcefile(function))
        installed.append(
            {
                "operation": label,
                "module": function.__module__,
                "name": function.__qualname__,
                "source": str(source),
                "source_sha256": hashlib.sha256(source.read_bytes()).hexdigest(),
            }
        )
        setattr(owner, name, wrapped(function, label, **options))

    for owner, name, label in (
        (stock_df, "_cholesky_eri", "raw_and_whitened_factor_construction"),
        (stock_df, "_decompose_j2c", "metric_decomposition_and_transform"),
        (stock_df, "cholesky", "metric_cholesky"),
        (stock_df, "solve_triangular", "metric_triangular_solve"),
        (int3c2e_bdiv, "int2c2e", "raw_metric"),
        (df_jk, "get_jk", "logical_jk"),
        (df_jk, "factorize_dm", "density_factorization"),
        (df_jk, "contract", "jk_tensor_contract"),
        (hf.SCF, "eig", "eigensystem"),
        (diis.CDIIS, "update", "diis"),
        (hf_grad, "_hcore_energy", "one_electron_force"),
        (df_grad, "_jk_energy_per_atom", "df_force_response"),
        (df_grad, "contract", "response_tensor_contract"),
        (df_grad, "_gen_metric_solver", "response_metric_solver"),
        (df_grad, "cholesky", "response_metric_cholesky"),
        (df_grad, "eigh", "response_metric_eigensystem"),
    ):
        install(owner, name, label)
    install(stock_df.DF, "build", "df_build", factor_owner=True)
    install(
        int3c2e_bdiv.Int3c2eOpt,
        "int3c2e_evaluator",
        "raw_evaluator_setup",
        evaluator=True,
    )

    systems = scaled_geometries(case.atoms, args.batch)
    assert [
        [
            {"element": element, "coordinates_bohr": list(position)}
            for element, position in atoms
        ]
        for atoms in systems
    ] == workload["geometries"]
    engines = []
    for atoms in systems:
        mol = gto.M(
            atom=atoms,
            unit="Bohr",
            basis=case.pyscf_basis,
            cart=case.basis_representation == "cartesian",
            charge=case.charge,
            spin=case.multiplicity - 1,
            verbose=0,
        )
        engine = uhf.UHF(mol) if case.method == "uhf" else scf.RHF(mol)
        engine = engine.density_fit(auxbasis=case.pyscf_basis).to_gpu()
        engine.conv_tol = workload["energy_tolerance"]
        engine.conv_tol_grad = workload["reference_gradient_tolerance"]
        engine.direct_scf_tol = 1e-14
        engine.max_cycle = 100
        engine.kernel()
        engine.nuc_grad_method().kernel()
        assert engine.converged
        engines.append(engine)
    densities = [engine.make_rdm1().copy() for engine in engines]
    _gpu_sample(engines, densities, cp, -1, True)
    density_hashes = [
        hashlib.sha256(cp.asnumpy(d).tobytes()).hexdigest() for d in densities
    ]

    def memory():
        raw = subprocess.check_output(
            [
                "nvidia-smi",
                "--query-compute-apps=pid,used_memory",
                "--format=csv,noheader,nounits",
            ],
            text=True,
        )
        entries = [line.split(",") for line in raw.splitlines()]
        memory_bytes = sum(
            int(mib) * 1024**2 for pid, mib in entries if int(pid) == os.getpid()
        )
        return [
            time.monotonic_ns(),
            memory_bytes,
            int(Path("/proc/self/statm").read_text().split()[1])
            * os.sysconf("SC_PAGE_SIZE"),
        ]

    samples, sampling_errors = [], []
    stop = threading.Event()

    def sample():
        try:
            while not stop.is_set():
                samples.append(memory())
                stop.wait(0.05)
        except (OSError, ValueError, subprocess.CalledProcessError) as error:
            sampling_errors.append(repr(error))

    payload = {
        "scope": __doc__,
        "case": args.case,
        "batch": args.batch,
        "slurm_job_id": os.environ["SLURM_JOB_ID"],
        "cuda_visible_devices": os.environ.get("CUDA_VISIBLE_DEVICES"),
        "runner_sha256": hashlib.sha256(Path(__file__).read_bytes()).hexdigest(),
        "reference_sha256": hashlib.sha256(args.reference.read_bytes()).hexdigest(),
        "installed_wrappers": installed,
        "operations": rows,
        "versions": {
            name: importlib.metadata.version(name)
            for name in (
                "gpu4pyscf-cuda12x",
                "cupy-cuda12x",
                "cutensor-cu12",
                "pyscf",
                "numpy",
            )
        },
        "factor_observations": factor_observations,
        "density_sha256": density_hashes,
        "pool_before": {
            "used_bytes": cp.get_default_memory_pool().used_bytes(),
            "retained_bytes": cp.get_default_memory_pool().total_bytes(),
        },
    }
    worker = threading.Thread(target=sample)
    try:
        samples.append(memory())
        worker.start()
        enabled = True
        cp.cuda.profiler.start()
        try:
            result = _gpu_sample(engines, densities, cp, 0, True)
        finally:
            cp.cuda.profiler.stop()
            enabled = False
            stop.set()
            worker.join()
        samples.append(memory())
        payload.update(
            result=result,
            process_samples=samples,
            sampling_errors=sampling_errors,
            sampled_device_peak=max(row[1] for row in samples),
            sampled_host_peak=max(row[2] for row in samples),
        )
        clean = reference["gpu4pyscf"]["warm_samples"][0]
        payload["pool_after"] = {
            "used_bytes": cp.get_default_memory_pool().used_bytes(),
            "retained_bytes": cp.get_default_memory_pool().total_bytes(),
        }
        payload["clean_reference_convergence"] = clean["convergence"]
        payload["same_scf_iterations_as_clean"] = [
            item["iterations"] for item in clean["convergence"]
        ] == [item["iterations"] for item in result["convergence"]]
        energy_error = float(
            np.max(
                np.abs(np.array(result["energies_hartree"]) - clean["energies_hartree"])
            )
        )
        force_error = float(
            np.max(
                np.abs(
                    np.array(result["forces_hartree_per_bohr"])
                    - clean["forces_hartree_per_bohr"]
                )
            )
        )
        payload.update(
            energy_error_from_clean_stock=energy_error,
            force_error_from_clean_stock=force_error,
        )
        assert all(row["converged"] for row in result["convergence"])
        assert [
            hashlib.sha256(cp.asnumpy(d).tobytes()).hexdigest() for d in densities
        ] == density_hashes
        assert not sampling_errors and rows and factor_observations
        assert energy_error <= gates["maximum_energy_error_hartree"]
        assert force_error <= gates["maximum_force_error_hartree_per_bohr"]
    except Exception as error:
        payload["failure"] = repr(error)
        raise
    finally:
        args.output.write_text(json.dumps(payload, indent=2) + "\n")


if __name__ == "__main__":
    main()
