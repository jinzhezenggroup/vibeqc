"""Retain the current external comparison, including every failed admission.

This collector does not rerun physics, change thresholds, or turn unmatched
convergence branches into fixed-work claims. Missing completion requirements
remain explicit even when a subset of endpoints beats the reference.
"""

from __future__ import annotations

import argparse
import collections
import hashlib
import json
import statistics
from pathlib import Path


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--output", type=Path, required=True)
    args = parser.parse_args()
    args.output.mkdir(parents=True, exist_ok=False)
    root = Path(".artifacts/issue206")
    manifest = {"scope": __doc__, "files": [], "incomplete": []}

    def retain(path, relative, *, value=None):
        original = path.read_bytes()
        data = (
            original if value is None else (json.dumps(value, indent=2) + "\n").encode()
        )
        if len(data) > 1 << 20 and path.suffix == ".json":
            data = (json.dumps(json.loads(data), separators=(",", ":")) + "\n").encode()
        assert len(data) <= 1 << 20, str(path)
        target = args.output / relative
        target.parent.mkdir(parents=True, exist_ok=True)
        target.write_bytes(data)
        manifest["files"].append(
            {
                "path": relative,
                "sha256": sha(target),
                "bytes": len(data),
                "original_path": str(path),
                "original_sha256": sha(path),
            }
        )

    summary = {
        "scope": "Ordinary converged latency with engine-local immutable post-cold D and equivalent untimed priming; no cross-engine fixed-work claim.",
        "warm": [],
        "rebuild": [],
        "all_mandatory_acceptance_passed": False,
    }
    for directory in ("fresh-warm-v2", "fresh-warm-remaining"):
        campaign = root / directory
        retain(campaign / "manifest.json", f"campaigns/{directory}.json")
        for path in sorted(campaign.glob("*.json")):
            if path.name == "manifest.json":
                continue
            data = json.loads(path.read_text())
            relative = f"{directory}/{path.name}"
            retain(path, relative)
            if "workload" not in data:
                continue
            paired = data["accuracy"]["paired_warm_repeats"]
            gates = data["settings"]["gates"]
            assert len(paired) == 7
            energy = max(row["maximum_energy_error_hartree"] for row in paired)
            force = max(
                (row["maximum_force_error_hartree_per_bohr"] or 0) for row in paired
            )
            numerics = energy <= gates["maximum_energy_error_hartree"] and (
                gates["maximum_force_error_hartree_per_bohr"] is None
                or force <= gates["maximum_force_error_hartree_per_bohr"]
            )
            medians, dispersion, branches = {}, {}, {}
            for engine in ("vibeqc", "gpu4pyscf"):
                samples = data[engine]["warm_samples"]
                assert len(samples) == 7
                assert all(c["converged"] for s in samples for c in s["convergence"])
                times = [s["seconds"] for s in samples]
                median = statistics.median(times)
                medians[engine] = median
                dispersion[engine] = (
                    statistics.median(abs(t - median) for t in times) / median
                )
                branches[engine] = dict(
                    collections.Counter(
                        str([c["iterations"] for c in s["convergence"]])
                        for s in samples
                    )
                )
            summary["warm"].append(
                {
                    "path": relative,
                    "case": data["workload"]["case"],
                    "aos": data["workload"]["ao_count"],
                    "batch": data["workload"]["batch_size"],
                    "properties": data["workload"]["properties"],
                    "gates": gates,
                    "all_seven_numerical_pairs_passed": numerics,
                    "original_gate": data["gate"],
                    "maximum_energy_error": energy,
                    "maximum_force_error": force,
                    "median_seconds": medians,
                    "descriptive_relative_median_absolute_deviation": dispersion,
                    "ordinary_stock_over_native_ratio": medians["gpu4pyscf"]
                    / medians["vibeqc"],
                    "update_distributions": branches,
                    "metric": data["settings"]["density_fitting_metric_diagnostics"],
                }
            )
    retain(root / "diagnose-96-batch4/report.json", "diagnosis/96-batch4.json")
    retain(root / "rebuild-smoke/manifest.json", "campaigns/rebuild-smoke.json")
    for phase in ("cold", "changed"):
        directory = root / "rebuild-smoke" / phase
        path = directory / "manifest.json"
        retain(path, f"rebuild/192-b1-{phase}.json")
        retain(
            directory / "stock-provider-and-rank.json",
            f"rebuild/192-b1-{phase}-stock-rank.json",
        )
        data = json.loads(path.read_text())
        summary["rebuild"].append({"phase": phase, **data["summary"]})
        if phase == "changed":
            for engine in ("vibeqc", "gpu4pyscf"):
                retain(
                    directory / f"{engine}-seed.json",
                    f"rebuild/{engine}-changed-seed.json",
                )
    for name in ("manifest.json", "192-stock.json", "account.json"):
        retain(root / "stock-diagnostic-smoke" / name, f"stock-diagnostics/{name}")
    direct = root / "final-direct/manifest.json"
    if direct.exists() and json.loads(direct.read_text()).get("status") in (
        "passed",
        "failed",
    ):
        retain(direct, "direct/manifest.json")
        data = json.loads(direct.read_text())
        summary["final_direct"] = {"status": data["status"], "points": data["points"]}
        for path in sorted((root / "final-direct/matrix").glob("*.json")):
            retain(path, f"direct/{path.name}")
        failure_log = root / "final-direct/matrix.log"
        retain(
            failure_log,
            "direct/failure-log.json",
            value={
                "scope": "Exact output including failed cold-start tracebacks.",
                "output": failure_log.read_text(),
            },
        )
        diagnosis = root / "diagnose-direct-192"
        if (diagnosis / "manifest.json").exists():
            retain(diagnosis / "manifest.json", "direct/192-diagnosis-manifest.json")
            for label in ("final", "frozen"):
                path = diagnosis / f"{label}.log"
                retain(
                    path,
                    f"direct/192-{label}-diagnosis.json",
                    value=json.loads(path.read_text()),
                )
            sanitizer = diagnosis / "memcheck.log"
            if sanitizer.exists():
                retain(
                    sanitizer,
                    "direct/192-memcheck.json",
                    value={
                        "scope": "Intrusive diagnostic: zero sanitizer errors does not qualify the failed native calculation.",
                        "output": sanitizer.read_text(),
                    },
                )
    else:
        manifest["incomplete"].append(
            "Final rebuilt-library direct matrix is not complete."
        )
    # Missing evidence is not inferred from issue closure or from another size.
    manifest["incomplete"] += [
        "96-AO batch-4 strict warm force gate and 192-AO changed-geometry force gate fail.",
        "Complete cold/changed/warm-energy coverage for the other mandatory cells is still open.",
        "Stock/native complete operator and memory diagnostics are not available for every required cell.",
        "Practical unequal and constrained-memory external endpoint comparisons remain open.",
        "384/768-AO ordinary DF energy and force latency is slower than stock in this campaign.",
    ]
    for name in (
        "run-fresh-matrix.py",
        "run-remaining-matrix.py",
        "qualify-stock.py",
        "diagnose-96-batch4.py",
        "compare-rebuilds.py",
        "run-rebuild-smoke.py",
        "stock-warm-diagnostics.py",
        "run-stock-diagnostic-smoke.py",
        "run-final-direct.py",
        "diagnose-direct-192.py",
        "account-stock.py",
        "starting-information-audit.py",
    ):
        retain(root / name, f"reproduction/{name}.txt")
    for path in (
        Path("benchmarks/compare_gpu4pyscf_batch.py"),
        Path("benchmarks/_cases.py"),
    ):
        retain(path, f"reproduction/{path.name}.txt")
    retain(Path(__file__), "collector.py.txt")
    for job in (9852, 9853, 9855, 9858, 9860, 9861, 9862):
        path = Path(f".artifacts/issue409/terminal-observer/terminal-{job}.json")
        if path.exists():
            retain(path, f"campaigns/terminal-{job}.json")
    summary_path = args.output / "summary.json"
    summary_path.write_text(json.dumps(summary, indent=2) + "\n")
    manifest["derived_summary_sha256"] = sha(summary_path)
    (args.output / "manifest.json").write_text(json.dumps(manifest, indent=2) + "\n")
    print(
        "Retained",
        len(manifest["files"]),
        "files and",
        len(summary["warm"]),
        "warm cells; acceptance remains open.",
    )


if __name__ == "__main__":
    main()
